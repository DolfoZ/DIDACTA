  drev-ws-relay:
    image: python:3.11-slim
    container_name: drev-websocket-module
    restart: unless-stopped
    working_dir: /app
    # Instala dependencias y ejecuta el módulo WebSocket inline
    command: >
      sh -c "
        pip install --no-cache-dir websockets aiohttp aioredis &&
        python -c \"
        import os, json, asyncio, websockets, aioredis
        from aiohttp import web

        REDIS_URL = os.getenv('REDIS_URL', 'redis://192.168.88.11:6379/0')
        WS_PORT = int(os.getenv('WS_PORT', '8765'))
        BACKEND_API = os.getenv('BACKEND_API', 'http://192.168.88.20:8000')

        async def sync_telemetry():
            redis = await aioredis.from_url(REDIS_URL)
            pubsub = redis.pubsub()
            await pubsub.subscribe('drev:telemetry', 'drev:missions')
            async for msg in pubsub.listen():
                if msg['type'] == 'message':
                    payload = msg['data'].decode('utf-8')
                    for ws in list(connected_clients):
                        try: await ws.send(payload)
                        except: connected_clients.discard(ws)

        async def handle_client(websocket, path):
            connected_clients.add(websocket)
            try:
                async for message in websocket:
                    # Reenvía comandos del frontend al backend API
                    resp = await web.ClientSession().post(f'{BACKEND_API}/v1/ws-forward', data=message)
                    await websocket.send(await resp.text())
            except websockets.exceptions.ConnectionClosed:
                pass
            finally:
                connected_clients.discard(websocket)

        connected_clients = set()
        start_server = websockets.serve(handle_client, '0.0.0.0', WS_PORT)
        print(f'[DREV-WS] Módulo WebSocket activo en puerto {WS_PORT}')
        asyncio.gather(start_server, sync_telemetry())
        asyncio.get_event_loop().run_forever()
        \"
      "

      - REDIS_URL=redis://192.168.88.11:6379/0
      - WS_PORT=8765
      - BACKEND_API=http://192.168.88.20:8000
