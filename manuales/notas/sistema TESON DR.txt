    const nav = document.getElementById('nav');
    const contentArea = document.getElementById('content-area');
    const searchBox = document.getElementById('search');
    let activeSection = 'intro';

    function renderNav() {
      nav.innerHTML = MANUAL.map(m => `<li class="nav-item ${m.id===activeSection?'active':''}" data-id="${m.id}" onclick="loadSection('${m.id}')">${m.title}</li>`).join('');
    }

    function loadSection(id) {
      activeSection = id;
      renderNav();
      const section = MANUAL.find(s => s.id === id);
      if(!section) return;
      contentArea.innerHTML = `<div class="content active"><h2>${section.title}</h2>${section.content}</div>`;
      document.querySelector('.sidebar').classList.remove('open');
      window.scrollTo({top:0, behavior:'smooth'});
    }

    function filterManual(q) {
      const lowerQ = q.toLowerCase();
      nav.innerHTML = MANUAL.filter(m => m.title.toLowerCase().includes(lowerQ) || m.content.toLowerCase().includes(lowerQ))
        .map(m => `<li class="nav-item" data-id="${m.id}" onclick="loadSection('${m.id}')">${m.title}</li>`).join('');
      if(!q) renderNav();
    }

    function toggleTactical() {
      document.body.classList.toggle('tactical-mode');
      const isTactical = document.body.classList.contains('tactical-mode');
      document.querySelector('.btn').textContent = isTactical ? '🟡 Modo Normal' : '🔴 Modo Táctico';
    }

    function exportData() {
      const blob = new Blob([JSON.stringify(MANUAL, null, 2)], {type: 'application/json'});
      const a = document.createElement('a'); a.href = URL.createObjectURL(blob); a.download = 'manual_drev.json'; a.click();
    }

    // ⏱️ Reloj y estado
    setInterval(() => document.getElementById('clock').textContent = new Date().toLocaleString('es-ES'), 1000);

    // 🚀 Inicialización
    searchBox.addEventListener('input', e => filterManual(e.target.value));
    loadSection('intro');
  </script>
