sistema de comunicaciones 

Un Sistema de Comunicación Interoperativo es exactamente el paso natural que transforma tu proyecto de un CRUD administrativo a una plataforma de mando y control (C2) real. La interoperabilidad no es solo "que las apps se hablen", es que sistemas heterogéneos, agencias diferentes y dispositivos diversos intercambien datos en tiempo real, con seguridad certificada y sin perder contexto.
Aquí tienes una arquitectura técnica, estándares reales y un roadmap para hacerlo viable y profesional

function assignVehicle(mechanicId, vehicleId) {
    let assignments = DB.get('mechanic_vehicles');
    
    // Remove any existing assignment for this vehicle
    assignments = assignments.filter(a => a.vehicle_id !== vehicleId);
    
    assignments.push({
        mechanic_id: mechanicId,
        vehicle_id: vehicleId,
        assigned_at: new Date().toISOString()
    });
    DB.set('mechanic_vehicles', assignments);

    // Update vehicle state and mechanic reference
    let vehicles = DB.get('vehicles');
    const idx = vehicles.findIndex(v => v.id === vehicleId);
    if (idx !== -1) {
        vehicles[idx].id_mechanic = mechanicId;
        vehicles[idx].state = 'maintenance';
        DB.set('vehicles', vehicles);
    }

    showToast('Vehículo asignado al mecánico', 'success');
    closeModal();
    renderContent('mechanics');
    buildNavigation();
}

function openModal() {
    document.getElementById('modalOverlay').classList.add('active');
    document.body.style.overflow = 'hidden';
}

function closeModal(event) {
    if (event && event.target !== event.currentTarget) return;
    document.getElementById('modalOverlay').classList.remove('active');
    document.body.style.overflow = '';
}

// --- TOAST NOTIFICATIONS ---
function showToast(message, type = 'info') {
    const container = document.getElementById('toastContainer');
    const toast = document.createElement('div');
    toast.className = `toast toast-${type}`;
    toast.textContent = message;
    container.appendChild(toast);
    setTimeout(() => toast.remove(), 3000);
}

const GRID_OFFSET_X = 40;
const GRID_OFFSET_Y = 40;
const GRID_SCALE = 0.08;

function init() {
  renderHierarchy();
  renderBatteryStatus();
  renderGridMarkers();
  startClock();
  startSimulation();
  setupGridInteraction();
  addLog('sys', 'SISTEMA SANTA BÁRBARA INICIALIZADO');
  addLog('sys', 'Canales seguros verificados. Protocolo EDXL-CAP activo.');
  addLog('enc', `Cifrado activo: ${STATE.enc}`);
  addLog('sys', 'Esperando designación de objetivos...');

function startClock() {
  const el = document.getElementById('clock');
  setInterval(() => {
    const now = new Date();
    el.textContent = now.toISOString().split('T')[1].split('.')[0] + ' Z';
  }, 1000);
}

function renderHierarchy() {
  const tree = document.getElementById('hierarchyTree');
  tree.innerHTML = '';
  const order = ['cp', 'fo1', 'fo2', 'bat1', 'bat2', 'log'];
  let online = 0;
  order.forEach(id => {
    const u = STATE.units[id];
    if(u.status === 'ONLINE') online++;
    const dotClass = u.status === 'ONLINE' ? 'dot-green' : u.status === 'DEGRADED' ? 'dot-amber' : 'dot-red';
    const div = document.createElement('div');
    div.className = `tree-item ${STATE.role === id.slice(0,2) ? 'active' : ''}`;
    div.innerHTML = `<span class="role">${u.role}</span> ${u.name} <span class="status"><span class="status-dot ${dotClass}"></span>${u.status}</span>`;
    div.onclick = () => {
      document.querySelectorAll('.tree-item').forEach(i => i.classList.remove('active'));
      div.classList.add('active');
    };
    tree.appendChild(div);
  });
  document.getElementById('onlineCount').textContent = `${online}/6`;
}

function renderBatteryStatus() {
  const cont = document.getElementById('batteryStatus');
  cont.innerHTML = '';
  ['bat1', 'bat2'].forEach(id => {
    const b = STATE.units[id];
    const cls = b.status === 'READY' ? 'ready' : b.status === 'FIRING' ? 'firing' : b.status === 'RELOAD' ? 'reload' : 'maint';
    const div = document.createElement('div');
    div.className = `bat-card ${cls}`;
    div.innerHTML = `<div class="bat-name">${b.name}</div><div class="bat-guns">${b.guns} U</div><div class="bat-ammunition">Munición: ${b.ammo}%</div>`;
    cont.appendChild(div);
  });
}

function renderGridMarkers() {
  const grid = document.getElementById('tacticalGrid');
  grid.innerHTML = '';
  Object.values(STATE.units).forEach(u => {
    if(!u.grid) return;
    const coords = u.grid.match(/\d+/g);
    if(!coords || coords.length < 2) return;
    const x = parseInt(coords[0]) * GRID_SCALE + GRID_OFFSET_X;
    const y = parseInt(coords[1]) * GRID_SCALE + GRID_OFFSET_Y;
    const marker = document.createElement('div');
    marker.className = `grid-marker ${u.color}`;
    marker.style.left = x + '%';
    marker.style.top = y + '%';
    marker.title = `${u.name}\nEstado: ${u.status}\nGrid: ${u.grid}`;
    grid.appendChild(marker);
  });
}

function setupGridInteraction() {
  const grid = document.getElementById('tacticalGrid');
  grid.addEventListener('click', (e) => {
    if(STATE.role !== 'fo') {
      addLog('warn', 'ACCESO DENEGADO: Solo observadores pueden designar objetivos.');
      return;
    }
    const rect = grid.getBoundingClientRect();
    const x = ((e.clientX - rect.left) / rect.width) * 100;
    const y = ((e.clientY - rect.top) / rect.height) * 100;
    const mgrsX = Math.floor((x - GRID_OFFSET_X) / GRID_SCALE);
    const mgrsY = Math.floor((y - GRID_OFFSET_Y) / GRID_SCALE);
    if(mgrsX < 100 || mgrsY < 100) return;
    const gridRef = `NK ${mgrsX} ${mgrsY}`;
    createTarget(gridRef, x, y);
    requestFireMission(gridRef);
  });
}

function createTarget(gridRef, x, y) {
  const grid = document.getElementById('tacticalGrid');
  const marker = document.createElement('div');
  marker.className = 'grid-marker marker-target';
  marker.style.left = x + '%';
  marker.style.top = y + '%';
  grid.appendChild(marker);
  STATE.targets.push({ grid: gridRef, x, y });
  addLog('sys', `OBJETIVO DESIGNADO: ${gridRef}`);
}

function requestFireMission(targetGrid) {
  if(STATE.role !== 'fo') return;
  const id = 'FM-' + Math.floor(Math.random() * 9000 + 1000);
  const mission = {
    id, target: targetGrid, type: 'HE', rounds: 3, status: 'PENDING',
    priority: 'URGENT', requestedBy: STATE.units.fo1.name, timestamp: new Date()
  };
  STATE.missions.push(mission);
  renderMissionQueue();
  addLog('enc', `[FO] SOLICITUD DE FUEGO ${id} → ${targetGrid} | 3 RDS HE`);
  routeMessage('fo1', 'cp', `SOLICITUD FUEGO ${id} | GRID ${targetGrid} | PRIORIDAD URGENTE`);
}

function renderMissionQueue() {
  const q = document.getElementById('missionQueue');
  q.innerHTML = '';
  STATE.missions.slice(-5).reverse().forEach(m => {
    const div = document.createElement('div');
    div.className = `mission-card priority-${m.priority.toLowerCase()}`;
    let actions = '';
    if(STATE.role === 'cp' && m.status === 'PENDING') {
      actions = `<button class="btn-xs btn-approve" onclick="approveMission('${m.id}')">APROBAR</button>
                 <button class="btn-xs btn-reject" onclick="rejectMission('${m.id}')">RECHAZAR</button>`;
    } else if(m.status === 'APPROVED') {
      actions = `<span style="color:var(--warn); font-size:10px;">EN EJECUCIÓN</span>`;
    } else if(m.status === 'COMPLETE') {
      actions = `<span style="color:var(--primary); font-size:10px;">COMPLETADO</span>`;
    }
    div.innerHTML = `
      <div>
        <div style="font-weight:600;">${m.id} | ${m.target}</div>
        <div class="mission-meta">${m.type.toUpperCase()} | ${m.rounds} RDS | ${m.requestedBy}</div>
      </div>
      <div class="mission-actions">${actions}</div>
    `;
    q.appendChild(div);
  });
}

function approveMission(id) {
  const m = STATE.missions.find(x => x.id === id);
  if(!m) return;
  m.status = 'APPROVED';
  renderMissionQueue();
  addLog('sys', `MISIÓN ${id} APROBADA POR CP`);
  routeMessage('cp', 'bat1', `ORDEN DE FUEGO ${id} | GRID ${m.target} | ${m.rounds} RDS ${m.type} | PRIORIDAD URGENTE`);
  setTimeout(() => executeFire(m), 2000);
}

function rejectMission(id) {
  const m = STATE.missions.find(x => x.id === id);
  if(!m) return;
  m.status = 'REJECTED';
  renderMissionQueue();
  addLog('crit', `MISIÓN ${id} RECHAZADA`);
  routeMessage('cp', 'fo1', `MISIÓN ${id} RECHAZADA - REEVALUAR OBJETIVO`);
}

function executeFire(m) {
  const bat = STATE.units.bat1;
  bat.status = 'FIRING';
  renderBatteryStatus();
  addLog('sys', `BATERÍA 1 EN FUEGO -> ${m.target}`);
  setTimeout(() => {
    bat.status = 'RELOAD';
    bat.ammo = Math.max(10, bat.ammo - Math.floor(m.rounds * 2));
    renderBatteryStatus();
    addLog('sys', `IMPACTO CONFIRMADO EN ${m.target}`);
    m.status = 'COMPLETE';
    renderMissionQueue();
    routeMessage('bat1', 'cp', `EFECTO SOBRE OBJETIVO ${m.target} | MUNICIÓN RESTANTE ${bat.ammo}%`);
    setTimeout(() => { bat.status = 'READY'; renderBatteryStatus(); }, 3000);
  }, 2500);
}

function routeMessage(from, to, content) {
  if(STATE.jamming && Math.random() > 0.6) {
    addLog('crit', 'PERDIDA DE PAQUETE: Interferencia detectada en enlace');
    return;
  }
  const encContent = STATE.enc === 'PLAIN' ? content : `[${STATE.enc}] ${btoa(unescape(encodeURIComponent(content))).slice(0, 20)}...`;
  addLog('enc', `${STATE.units[from]?.name} → ${STATE.units[to]?.name}: ${encContent}`);
}

function sendMessage() {
  const input = document.getElementById('commInput');
  const txt = input.value.trim();
  if(!txt) return;
  const to = STATE.role === 'cp' ? 'ALL' : 'cp';
  routeMessage(STATE.role, to, txt);
  input.value = '';
}

document.getElementById('commInput').addEventListener('keypress', (e) => {
  if(e.key === 'Enter') sendMessage();
});

}

function renderMap(){
  const markers=document.getElementById('mapMarkers');markers.innerHTML='';
  const targets=document.getElementById('mapTargets');targets.innerHTML='';
  const lines=document.getElementById('mapLines');lines.innerHTML='';
  Object.values(S.units).forEach(u=>{
    const coords=u.grid.match(/\d+/g);
    if(!coords||coords.length<2)return;
    const x=(parseInt(coords[0])/1000)*90+5;
    const y=(parseInt(coords[1])/1000)*90+5;
    const mk=document.createElement('div');mk.className=`map-marker mk-${u.type}`;
    mk.style.left=x+'%';mk.style.top=y+'%';
    mk.innerHTML=`<div class="mk-inner"></div><div class="mk-label" style="color:${u.type==='cp'?'var(--primary)':u.type==='fo'?'var(--info)':u.type==='bat'?'var(--warn)':'var(--purple)'}">${u.callSign}</div>`;
    mk.title=`${u.name}\nMGRS: ${u.grid}\nEstado: ${u.status}`;
    mk.onmouseenter=showTooltip;mk.onmouseleave=hideTooltip;
    markers.appendChild(mk);
  });
  S.targets.forEach((t,i)=>{
    const coords=t.grid.match(/\d+/g);
    if(!coords||coords.length<2)return;
    const x=(parseInt(coords[0])/1000)*90+5;
    const y=(parseInt(coords[1])/1000)*90+5;
    const tgt=document.createElement('div');tgt.className='map-target';
    tgt.style.left=x+'%';tgt.style.top=y+'%';
    tgt.innerHTML=`<div class="tgt-inner"></div><div class="tgt-label">OBJ-${String(i+1).padStart(2,'0')} ${t.grid}</div>`;
    tgt.title=`Objetivo ${t.grid}\nTipo: ${t.type}\nEstado: ${t.status}`;
    targets.appendChild(tgt);
    // Fire line from nearest battery
    if(t.assignedBat){
      const bat=S.units[t.assignedBat];
      const bCoords=bat.grid.match(/\d+/g);
      if(bCoords){
        const bx=(parseInt(bCoords[0])/1000)*90+5;
        const by=(parseInt(bCoords[1])/1000)*90+5;
        const dist=Math.sqrt((x-bx)**2+(y-by)**2);
        const angle=Math.atan2(y-by,x-bx)*180/Math.PI;
        const fl=document.createElement('div');fl.className='fire-line';
        fl.style.left=bx+'%';fl.style.top=by+'%';fl.style.width=dist+'%';fl.style.transform=`rotate(${angle}deg)`;
        lines.appendChild(fl);
      }
    }
  });
}

function setupMap(){
  const map=document.getElementById('mapContainer');
  map.addEventListener('mousemove',e=>{
    const rect=map.getBoundingClientRect();
    const x=((e.clientX-rect.left)/rect.width)*100;
    const y=((e.clientY-rect.top)/rect.height)*100;
    const mgrsX=Math.floor(x*1000/90);
    const mgrsY=Math.floor(y*1000/90);
    document.getElementById('mgrsCursor').textContent=`NK ${String(mgrsX).padStart(3,'0')} ${String(mgrsY).padStart(3,'0')}`;
    const ch=document.getElementById('mapCrosshair');
    ch.style.display='block';ch.style.left=e.clientX-rect.left+'px';ch.style.top=e.clientY-rect.top+'px';
  });
  map.addEventListener('mouseleave',()=>{document.getElementById('mapCrosshair').style.display='none'});
  map.addEventListener('click',e=>{
    if(S.role!=='fo'){notify('Solo observadores avanzados pueden designar objetivos','n-warn');return}
    const rect=map.getBoundingClientRect();
    const x=((e.clientX-rect.left)/rect.width)*100;
    const y=((e.clientY-rect.top)/rect.height)*100;
    const mgrsX=Math.floor(x*1000/90);
    const mgrsY=Math.floor(y*1000/90);
    const gridRef=`NK ${String(mgrsX).padStart(3,'0')} ${String(mgrsY).padStart(3,'0')}`;
    createTarget(gridRef,x,y);
    requestFireMission(gridRef,x,y);
  });
}

function approveMission(id){
  const m=S.missions.find(x=>x.id===id);if(!m)return;
  m.status='APPROVED';
  // Assign to nearest available battery
  const bats=Object.values(S.units).filter(u=>u.type==='bat'&&u.status==='READY');
  if(bats.length>0){
    m.assignedBat=bats[0].id;
    bats[0].status='FIRING';
  }
  renderMissions();renderUnits();renderMap();
  addTerm('sys','CP',`MISIÓN ${m.id} APROBADA → Asignada a ${m.assignedBat?S.units[m.assignedBat].callSign:'—'}`);
  if(m.assignedBat){
    routeMsg('cp',m.assignedBat,`FIRE ORDER ${m.id} | GRID ${m.target} | ${m.rounds} RDS ${m.type} | METHOD: ${m.method} | PRIORITY: ${m.priority}`);
    setTimeout(()=>executeFire(m),2000);
  }
  notify(`Misión ${m.id} aprobada — Ejecución inminente`,'n-ok');
}

}

function executeFire(m){
  const bat=S.units[m.assignedBat];if(!bat)return;
  m.status='EXECUTING';renderMissions();
  addTerm('sys','CP',`🔥 FUEGO EN CURSO — ${bat.callSign} → ${m.target}`);
  routeMsg(m.assignedBat,'cp',`FIRING ${m.id} | ${bat.callSign} | ROUNDS AWAY`);
  // Simulate rounds
  let roundsFired=0;
  const fireInterval=setInterval(()=>{
    roundsFired++;
    bat.ammo=Math.max(0,bat.ammo-1);
    bat.ammoTypes.HE=Math.max(0,bat.ammoTypes.HE-1);
    addTerm('bat',bat.callSign.split('-')[0],`💥 RDS ${roundsFired}/${m.rounds} IMPACTANDO ${m.target}`);
    if(roundsFired>=m.rounds){
      clearInterval(fireInterval);
      m.status='COMPLETE';bat.status='RELOAD';
      renderMissions();renderUnits();renderAmmo();renderMap();renderTelemetry();
      addTerm('sys','CP',`✅ MISIÓN ${m.id} COMPLETADA — Efecto sobre objetivo confirmado`);
      routeMsg(m.assignedBat,'cp',`MISSION ${m.id} COMPLETE | ALL ROUNDS DELIVERED | EFFECT: ON TARGET`);
      notify(`Misión ${m.id} completada — ${m.rounds} rondas impactadas`,'n-ok');
      // Reload after delay
      setTimeout(()=>{bat.status='READY';bat.ammo=Math.min(100,bat.ammo+5);renderUnits();renderAmmo();},5000);
    }
  },1500);
}

  addTerm(from.type,from.callSign,`→ ${to.callSign}: ${enc}`);
  // Auto-reply for realism
  setTimeout(()=>{
    if(Math.random()>.3){
      const replies=['ACK','COPY ALL','ROGER','WILCO','STANDING BY','AFFIRMATIVE','CONFIRMED'];
      const reply=replies[Math.floor(Math.random()*replies.length)];
      addTerm(to.type,to.callSign,`← ${from.callSign}: ${reply} — ${encMsg('Mensaje recibido y procesado')}`);
    }
  },800+Math.random()*1500);
}

    function solvePhysics(dt) {
        const g = 9.80665;
        const speed = Math.sqrt(state[3]**2 + state[4]**2 + state[5]**2);
        if(speed < 1 && state[1] < 0) return;

        const rho = 1.225 * Math.exp(-state[1]/8500);
        const area = Math.PI * (0.077**2); 
        const drag = 0.5 * rho * speed * speed * 0.22 * area;
        const accDrag = drag / activeAmmo.mass;

        state[3] -= (state[3]/speed) * accDrag * dt;
        state[4] -= ((state[4]/speed) * accDrag + g) * dt;
        state[5] -= (state[5]/speed) * accDrag * dt;

        // Influencia del viento simulada por la IA
        state[3] += windVector.x * 0.01 * dt;
        state[5] += windVector.z * 0.01 * dt;

        if(heraActive && clock.getElapsedTime() < 12) {
            const thrust = 1300 / activeAmmo.mass; 
            state[4] += (state[4]/speed) * thrust * dt;
            state[5] += (state[5]/speed) * thrust * dt;
        }

        state[0] += state[3] * dt;
        state[1] += state[4] * dt;
        state[2] += state[5] * dt;
        if(state[1] > maxAltitude) maxAltitude = state[1];

        state[6] = -Math.atan2(state[4], state[5]);
        state[8] += state[11] * dt; 
    }

    private val messageQueue = mutableListOf<ByteArray>()

    init {
        // Configurar SSL para certificado pinning en producción

    }
    
    override fun onOpen(handshake: ServerHandshake?) {
        Timber.i("✓ WebSocket conectado: ${handshake?.httpStatus}")
        _connectionState.value = ConnectionState.Connected
        flushQueue()
    }
    
    override fun onMessage(message: String?) {
        message?.let { raw ->
            try {
                val decrypted = if (enableEncryption) decrypt(raw) else raw
                val tacticalMsg = parseMessage(decrypted)
                tacticalMsg?.let {
                    _messages.value = _messages.value + it
                }
            } catch (e: Exception) {
                Timber.e(e, "Error procesando mensaje")
            }
        }
    }
    
    override fun onClose(code: Int, reason: String?, remote: Boolean) {
        Timber.w("✗ WebSocket cerrado: $reason (code: $code, remote: $remote)")
        _connectionState.value = ConnectionState.Disconnected
    }
    
    override fun onError(ex: Exception?) {
        Timber.e(ex, "❌ Error WebSocket")
        _connectionState.value = ConnectionState.Error(ex ?: Exception("Unknown error"))
    }
    
    fun send(message: TacticalMessage, priority: Priority = Priority.ROUTINE) {
        try {
            val json = serializeMessage(message)
            val payload = if (enableEncryption) encrypt(json) else json
            
            val envelope = mapOf(
                "v" to "2.1",
                "p" to priority.name,
                "d" to payload,
                "ts" to System.currentTimeMillis()
            )
            // Utilizando Gson / Moshi u otra librería de serialización (simplificado)

            } else {

                Timber.d("⏳ En cola: ${message.type}")
            }
        } catch (e: Exception) {
            Timber.e(e, "Error enviando mensaje")
        }
    }
