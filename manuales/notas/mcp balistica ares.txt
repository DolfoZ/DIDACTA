@mcp.tool()
def get_projectile_info(projectile_id: str) -> Dict[str, Any]:
    """Retrieve detailed technical specifications for a specific artillery projectile."""
    pid = projectile_id.upper().replace(" ", "_")
    return PROJECTILES_DB.get(pid, {"error": "Projectile not found in database"})

@mcp.tool()
def get_charge_info(charge_id: str) -> Dict[str, Any]:
    """Retrieve specifications for a propelling charge (e.g., GB M3A1, WB M4A2)."""
    cid = charge_id.upper().replace(" ", "_")
    return CHARGES_DB.get(cid, {"error": "Charge not found in database"})

@mcp.tool()
def calculate_compatibility(projectile_id: str, charge_id: str) -> Dict[str, Any]:
    """Check if a projectile is compatible with a specific charge according to STANAG/Firing Tables."""
    pid = projectile_id.upper().replace(" ", "_")
    cid = charge_id.upper().replace(" ", "_")
    
    proj = PROJECTILES_DB.get(pid)
    charge = CHARGES_DB.get(cid)
    
    if not proj or not charge:
        return {"error": "Invalid projectile or charge ID"}
    
    is_compatible = pid in charge.get("compatible_projectiles", [])
    return {
        "compatible": is_compatible,
        "projectile": proj["name"],
        "charge": charge["designation"],
        "estimated_v0": charge["v0_mps"] if is_compatible else 0,
        "warning": "Check official firing tables (FT 155-AM-2) for operational use."
    }

@mcp.tool()
def get_drev_manual_content(query: str) -> str:
    """Search the DREV tactical manual for operational procedures and system specs."""
    # Simple keyword matching for the MCP context
    if "alineamiento" in query.lower() or "fine" in query.lower():
        return "Modos de Alineamiento: Fino (10min, máx precisión), Normal (4min, ±20m), Rápido (30s, ±0.7 mils)."
    elif "seguridad" in query.lower() or "limit" in query.lower():
        return "Límites de Seguridad: Tácticos (4x error probable), Mecánicos del arma. Rojo = Fuera de rango, Verde = Seguro."
    elif "ins" in query.lower():
        return "INS de fibra óptica, precisión 0.3 mil azimut, >200 Hz. Verificar oscilación antes de disparar."
    else:
        return DREV_MANUAL_SUMMARY

@mcp.tool()
def get_comms_hierarchy(role: str) -> Dict[str, Any]:
    """Get the role, clearance level, and access rights for a user in the communications network."""
    role_key = role.upper()
    if role_key in COMMS_HIERARCHY:
        return COMMS_HIERARCHY[role_key]
    return {"error": "Role not recognized"}
