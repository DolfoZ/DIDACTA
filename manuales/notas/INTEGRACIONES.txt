// Convierte las llamadas MCP (stdio) en API REST + WebSocket para el frontend
import { spawn } from 'child_process';
import express from 'express';
import { Server as WSServer } from 'ws';
import { EventEmitter } from 'events';

class MCPBridge extends EventEmitter {
  private ballisticProcess: any;
  private logisticsProcess: any;
  private requestId = 0;
  private pending = new Map<number, { resolve: Function; reject: Function }>();

  constructor() {
    super();
    this.startBallisticMCP();
    this.startLogisticsMCP();
  }

  private startBallisticMCP() {
    this.ballisticProcess = spawn('/opt/ares/venv/bin/python', [
      '/opt/ares/mcp/ares_ballistic_mcp_master.py'
    ], { stdio: ['pipe', 'pipe', 'pipe'] });

    let buffer = '';
    this.ballisticProcess.stdout.on('data', (data: Buffer) => {
      buffer += data.toString();
      const lines = buffer.split('\n');
      buffer = lines.pop() || '';
      for (const line of lines) {
        try {
          const msg = JSON.parse(line);
          if (msg.id && this.pending.has(msg.id)) {
            this.pending.get(msg.id)?.resolve(msg.result);
            this.pending.delete(msg.id);
          }
        } catch {}
      }
    });
  }

  private startLogisticsMCP() {
    this.logisticsProcess = spawn('/opt/ares/venv/bin/python', [
      '/opt/ares/mcp/ares_logistics_mcp.py'
    ], { stdio: ['pipe', 'pipe', 'pipe'] });

    let buffer = '';
    this.logisticsProcess.stdout.on('data', (data: Buffer) => {
      buffer += data.toString();
      const lines = buffer.split('\n');
      buffer = lines.pop() || '';
      for (const line of lines) {
        try {
          const msg = JSON.parse(line);
          if (msg.id && this.pending.has(msg.id)) {
            this.pending.get(msg.id)?.resolve(msg.result);
            this.pending.delete(msg.id);
          }
        } catch {}
      }
    });
  }

  async callBallisticTool(tool: string, args: any): Promise<any> {
    const id = ++this.requestId;
    const request = { jsonrpc: '2.0', id, method: 'tools/call', params: { name: tool, arguments: args } };
    this.ballisticProcess.stdin.write(JSON.stringify(request) + '\n');
    return new Promise((resolve, reject) => {
      this.pending.set(id, { resolve, reject });
      setTimeout(() => { this.pending.delete(id); reject(new Error('Timeout')); }, 15000);
    });
  }

  async callLogisticsTool(tool: string, args: any): Promise<any> {
    const id = ++this.requestId;
    const request = { jsonrpc: '2.0', id, method: 'tools/call', params: { name: tool, arguments: args } };
    this.logisticsProcess.stdin.write(JSON.stringify(request) + '\n');
    return new Promise((resolve, reject) => {
      this.pending.set(id, { resolve, reject });
      setTimeout(() => { this.pending.delete(id); reject(new Error('Timeout')); }, 15000);
    });
  }
}

export const mcpBridge = new MCPBridge();

// Express router para exponer los MCP como API REST
const router = express.Router();

// Balística
router.post('/ballistic/solve-fire-mission', async (req, res) => {
  const result = await mcpBridge.callBallisticTool('solve_fire_mission', req.body);
  res.json(result);
});
