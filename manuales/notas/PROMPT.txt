import numpy as np
from src.utils.constants import P0, T0, R_SPECIFIC, LAPSE_RATE, TROPOPAUSE_HEIGHT, T_TROPOPAUSE, G0, GAMMA_AIR

class MolecularAir:
    def __init__(self):
        pass
    
    def get_properties(self, altitude):
        if altitude <= TROPOPAUSE_HEIGHT:
            T = T0 + LAPSE_RATE * altitude
            P = P0 * (T / T0) ** (-G0 / (LAPSE_RATE * R_SPECIFIC))
        else:
            T = T_TROPOPAUSE
            P_trop = P0 * (T_TROPOPAUSE / T0) ** (-G0 / (LAPSE_RATE * R_SPECIFIC))
            P = P_trop * np.exp(-G0 * (altitude - TROPOPAUSE_HEIGHT) / (R_SPECIFIC * T))
        rho = P / (R_SPECIFIC * T)
        a = np.sqrt(GAMMA_AIR * R_SPECIFIC * T)
        return {'temperature': T, 'pressure': P, 'density': rho, 'speed_of_sound': a}

import numpy as np
from src.utils.constants import INCH_TO_METER

class ProjectileGeometry:
    def __init__(self, caliber, unit='mm'):
        if unit == 'mm':
            self.caliber_m = caliber / 1000.0
        elif unit == 'inch':
            self.caliber_m = caliber * INCH_TO_METER
        else:
            self.caliber_m = caliber
    
    def cross_sectional_area(self):
        return np.pi * (self.caliber_m / 2) ** 2

import numpy as np
from scipy.interpolate import CubicSpline

class DragCoefficient:
    G1_TABLE = np.array([
        [0.0, 0.2629], [0.4, 0.2629], [0.6, 0.2720], [0.8, 0.3610],
        [0.9, 0.4750], [1.0, 0.6800], [1.1, 0.6900], [1.2, 0.6150],
        [1.4, 0.4950], [1.6, 0.4200], [1.8, 0.3730], [2.0, 0.3420],
        [2.5, 0.2880], [3.0, 0.2510], [4.0, 0.2050], [5.0, 0.1750]
    ])
    
    G7_TABLE = np.array([
        [0.0, 0.1198], [0.5, 0.1198], [0.8, 0.1250], [0.9, 0.1580],
        [1.0, 0.3200], [1.1, 0.3900], [1.2, 0.3550], [1.4, 0.2970],
        [1.6, 0.2600], [1.8, 0.2350], [2.0, 0.2170], [2.5, 0.1810],
        [3.0, 0.1650], [4.0, 0.1350], [5.0, 0.1160]
    ])
    
    TABLES = {'G1': G1_TABLE, 'G7': G7_TABLE}
    
    def __init__(self, drag_function='G1', ballistic_coefficient=1.0):
        if drag_function not in self.TABLES:
            raise ValueError(f"Funcion no soportada: {drag_function}. Use G1 o G7")
        self.table = self.TABLES[drag_function]
        self.bc = ballistic_coefficient
        self.spline = CubicSpline(self.table[:, 0], self.table[:, 1], extrapolate=True)
    
    def get_cd(self, mach):
        mach = np.clip(mach, self.table[0, 0], self.table[-1, 0])
        return float(self.spline(mach)) / self.bc
EOF

from src.utils.constants import G0, EARTH_RADIUS

class Gravity:
    def __init__(self, model='constant'):
        self.model = model
    
    def get_gravity(self, altitude):
        if self.model == 'constant':
            return G0
        else:
            return G0 * (EARTH_RADIUS / (EARTH_RADIUS + altitude)) ** 2

import numpy as np
from src.utils.constants import EARTH_ANGULAR_VELOCITY, DEG_TO_RAD

class Coriolis:
    def __init__(self, latitude=45.0):
        self.latitude_rad = latitude * DEG_TO_RAD
        self.omega = np.array([
            EARTH_ANGULAR_VELOCITY * np.cos(self.latitude_rad),
            0.0,
            -EARTH_ANGULAR_VELOCITY * np.sin(self.latitude_rad)
        ])
    
    def get_coriolis_acceleration(self, velocity):
        return -2.0 * np.cross(self.omega, velocity)

import numpy as np
from scipy.spatial.transform import Rotation
from src.utils.constants import DEG_TO_RAD

class RotationTransform:
    def __init__(self, azimuth=0.0, elevation=0.0):
        az_rad = azimuth * DEG_TO_RAD
        el_rad = elevation * DEG_TO_RAD
        r_az = Rotation.from_euler('z', -az_rad)
        r_el = Rotation.from_euler('y', el_rad)
        self.rotation = r_el * r_az
    
    def muzzle_to_ned(self, muzzle_velocity):
        v_muzzle = np.array([muzzle_velocity, 0.0, 0.0])
        return self.rotation.apply(v_muzzle)

import numpy as np
from src.atmosphere.molecular_air import MolecularAir
from src.geophysics.gravity import Gravity

class Forces:
    def __init__(self, drag_model, geometry, mass, coriolis_model=None):
        self.drag_model = drag_model
        self.geometry = geometry
        self.mass = mass
        self.atmosphere = MolecularAir()
        self.gravity_model = Gravity()
        self.coriolis_model = coriolis_model
        self.area = geometry.cross_sectional_area()
    
    def calculate(self, t, state):
        pos = state[0:3]
        vel = state[3:6]
        altitude = -pos[2]
        speed = np.linalg.norm(vel)
        
        if speed < 1e-10:
            return np.zeros(3)
        
        atm = self.atmosphere.get_properties(altitude)
        mach = speed / atm['speed_of_sound']
        cd = self.drag_model.get_cd(mach)
        drag_force = -0.5 * atm['density'] * speed * vel * self.area * cd
        
        gravity = self.gravity_model.get_gravity(altitude)
        gravity_force = np.array([0.0, 0.0, self.mass * gravity])
        
        total_force = drag_force + gravity_force
        
        if self.coriolis_model is not None:
            coriolis_acc = self.coriolis_model.get_coriolis_acceleration(vel)
            total_force += self.mass * coriolis_acc
        
        return total_force
EOF

from scipy.integrate import solve_ivp

class RKF45:
    def __init__(self, derivatives_func):
        self.derivatives = derivatives_func
    
    def integrate(self, initial_state, t_span, t_eval=None):
        result = solve_ivp(self.derivatives, t_span, initial_state, 
                          method='DOP853', t_eval=t_eval, rtol=1e-10, atol=1e-12)
        return {'t': result.t, 'y': result.y}

import numpy as np
import pandas as pd
from src.dynamics.integrator_rkf45 import RKF45

class TrajectoryGenerator:
    def __init__(self, force_calculator, mass):
        self.force_calc = force_calculator
        self.mass = mass
    
    def derivatives(self, t, y):
        vel = y[3:6]
        forces = self.force_calc.calculate(t, y)
        acc = forces / self.mass
        return np.concatenate([vel, acc])
    
    def generate(self, initial_state, t_max, dt=0.1):
        t_eval = np.arange(0, t_max, dt)
        integrator = RKF45(self.derivatives)
        result = integrator.integrate(initial_state, (0, t_max), t_eval)
        df = pd.DataFrame({
            'time': result['t'],
            'x': result['y'][0], 'y': result['y'][1], 'z': result['y'][2],
            'vx': result['y'][3], 'vy': result['y'][4], 'vz': result['y'][5]
        })
        return df

import pandas as pd
import numpy as np

class TableBuilder:
    @staticmethod
    def build_range_table(trajectories):
        table = []
        for elev, traj in trajectories.items():
            impact = traj.iloc[-1]
            table.append({
                'elevation': elev,
                'range': np.sqrt(impact['x']**2 + impact['y']**2),
                'time': impact['time']
            })
        return pd.DataFrame(table)
    
    @staticmethod
    def export_csv(df, filename):
        df.to_csv(filename, index=False)

import pandas as pd
from src.atmosphere.molecular_air import MolecularAir

class AtmosphereComparator:
    def __init__(self):
        self.isa = MolecularAir()
    
    def compare_with_data(self, altitudes, measured_densities):
        results = []
        for h, rho_measured in zip(altitudes, measured_densities):
            props = self.isa.get_properties(h)
            results.append({
                'altitude': h,
                'isa_density': props['density'],
                'measured_density': rho_measured,
                'difference': props['density'] - rho_measured
            })
        return pd.DataFrame(results)
