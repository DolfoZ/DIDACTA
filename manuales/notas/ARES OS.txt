}));

// ==================== PHYSICS ENGINE (JS) ====================
const PhysicsEngine = {
  G0: 9.80665, R_AIR: 287.05, GAMMA: 1.4, OMEGA: 7.2921159e-5, RADIO_T: 6371000,
  LAT: 14.08 * Math.PI / 180,
  G7_TABLE: [[0,0.12],[0.7,0.122],[0.85,0.135],[0.95,0.2],[1,0.35],[1.05,0.38],[1.1,0.375],[1.2,0.355],[1.5,0.31],[2,0.26],[3,0.19],[5,0.15]],
  atmosfera(h) {
    if (h <= 11000) { let T = 288.15 - 0.0065*h; let P = 101325 * Math.pow(T/288.15, 5.256); return [P/(this.R_AIR*T), T]; }
    else { let P = 22632 * Math.exp(-this.G0*(h-11000)/(this.R_AIR*216.65)); return [P/(this.R_AIR*216.65), 216.65]; }
  },
  gravedad(h) { return this.G0 * Math.pow(this.RADIO_T/(this.RADIO_T+h), 2); },
  g7cd(mach) {
    let t = this.G7_TABLE;
    if (mach <= t[0][0]) return t[0][1];
    if (mach >= t[t.length-1][0]) return t[t.length-1][1];
    for (let i = 0; i < t.length-1; i++) {
      if (mach >= t[i][0] && mach <= t[i+1][0]) {
        let f = (mach - t[i][0]) / (t[i+1][0] - t[i][0]);
        return t[i][1] + f * (t[i+1][1] - t[i][1]);
      }
    }
    return 0.25;
  },
  simulate(proj, qeMils, vzW = 0) {
    let theta = qeMils * 0.001;
    let v0 = proj.v0;
    let x = 0, y = 0.5, z = 0, vx = v0*Math.cos(theta), vy = v0*Math.sin(theta), vz = 0;
    let dt = 0.05, t = 0, maxY = 0, points = [];
    while (y >= 0 && t < 300) {
      let [rho, temp] = this.atmosfera(y);
      let vr = Math.sqrt(vx**2 + vy**2 + (vz-vzW)**2) || 1e-6;
      let mach = vr / Math.sqrt(this.GAMMA * this.R_AIR * temp);
      let cd = this.g7cd(mach);
      let area = Math.PI * (proj.caliber/2000)**2;
      let kDrag = 0.5 * rho * cd * area / proj.mass;
      let g = this.gravedad(y);
      let corX = 2*this.OMEGA*(vy*Math.sin(this.LAT) - vz*Math.cos(this.LAT));
      let corY = -2*this.OMEGA*vx*Math.sin(this.LAT);
      let corZ = 2*this.OMEGA*vx*Math.cos(this.LAT);
      let ax = -kDrag*vr*vx + corX, ay = -kDrag*vr*vy - g + corY, az = -kDrag*vr*(vz-vzW) + corZ;
      x += vx*dt; y += vy*dt; z += vz*dt; vx += ax*dt; vy += ay*dt; vz += az*dt; t += dt;
      if (y > maxY) maxY = y;
      if (Math.floor(t*10) % 5 === 0) points.push({t: +t.toFixed(2), x: +x.toFixed(1), y: +y.toFixed(1), z: +z.toFixed(2), v: +Math.sqrt(vx**2+vy**2+vz**2).toFixed(1) });
    }
    if (y < 0) { let frac = -y/vy; x += vx*frac; z += vz*frac; t -= dt-frac; }
    return { range: x, deflection: z, time: t, maxAlt: maxY, points };
  },
  inverse(proj, targetRange, tol = 1) {
    let lo = 50, hi = 1600;
    for (let i = 0; i < 60; i++) {
      let mid = (lo+hi)/2;
      let r = this.simulate(proj, mid).range;
      if (Math.abs(r - targetRange) < tol) return mid;
      if (r < targetRange) lo = mid; else hi = mid;
    }
    return (lo+hi)/2;
  }
};

// ==================== TERMINAL APP ====================
function TerminalApp({ winId }) {
  const [output, setOutput] = useState([
    { text: 'ARES TERMINAL v2.6 — Santa Barbara Root Shell', color: '#2dd4bf' },
    { text: 'Escribe "help" para comandos disponibles.', color: '#8b949e' },
  ]);
  const [input, setInput] = useState('');
  const [history, setHistory] = useState([]);
  const [histIdx, setHistIdx] = useState(-1);
  const outRef = useRef(null);

  useEffect(() => { outRef.current?.scrollTo(0, outRef.current.scrollHeight); }, [output]);

  const execute = (cmd) => {
    const args = cmd.trim().split(/\s+/);
    const c = args[0]?.toLowerCase();
    let result = '';
    if (c === 'help') result = 'Comandos: help, clear, date, whoami, ls, pwd, cat, drev [proy] [carga] [rango], status, proxmox, net, calc [expr], echo [text]';
    else if (c === 'clear') { setOutput([]); return; }
    else if (c === 'date') result = new Date().toISOString();
    else if (c === 'whoami') result = 'Santa Barbara (root)';
    else if (c === 'ls') result = 'home/  etc/  opt/  var/  mnt/  tmp/';
    else if (c === 'pwd') result = '/home/santa-barbara';
    else if (c === 'cat' && args[1]) result = `[Contenido de ${args[1]} simulado]`;
    else if (c === 'status') result = 'ARES OS v2.6 | Kernel: Linux 6.8.0-ares | Uptime: 3h 42m | CPU: 12% | RAM: 4.2/16GB | Proxmox GW: 192.168.88.1 | VPN: WireGuard Active | DNS: 1.1.1.1';
    else if (c === 'proxmox') result = 'Clúster Proxmox 192.168.88.1: 3 nodos online (pve1, pve2, pve3), 12 VMs activas';
    else if (c === 'net') result = 'eth0: 192.168.88.10/24 | GW: 192.168.88.1 | DNS: 1.1.1.1 | wg0: 10.0.0.1/24';
    else if (c === 'drev' && args[1]) {
      let proj = MUNITIONS_DB[args[1]];
      if (!proj) result = 'Proyectil no encontrado. Use: 155_HE_M107, 155_HERA_M549A1, 105_HE_M1';
      else {
        let carga = args[2] || Object.keys(proj.charges)[0];
        let rango = parseFloat(args[3]) || 10000;
        if (!proj.charges[carga]) carga = Object.keys(proj.charges)[0];
        let p = { ...proj, v0: proj.charges[carga] };
        let sim = PhysicsEngine.simulate(p, PhysicsEngine.inverse(p, rango));
        result = `DREV: ${proj.name} + ${carga} | QE: ${PhysicsEngine.inverse(p,rango).toFixed(1)} mils | Alcance: ${sim.range.toFixed(0)}m | TOF: ${sim.time.toFixed(1)}s | Deriva: ${sim.deflection.toFixed(2)}m`;
      }
    } else if (c === 'calc' && args[1]) {
      try { result = eval(args.slice(1).join(' ')).toString(); } catch { result = 'Error'; }
    } else if (c === 'echo') result = args.slice(1).join(' ');
    else if (c) result = `Comando "${c}" no reconocido. Escribe "help" para ayuda.`;
    setOutput(prev => [...prev, { text: `root@ares:~$ ${cmd}`, color: '#2dd4bf' }, ...(result ? [{ text: result, color: '#e6edf3' }] : [])]);
    if (cmd.trim()) setHistory(prev => [...prev, cmd.trim()]);
    setHistIdx(-1);
  };

  const handleKey = (e) => {
    if (e.key === 'Enter') { execute(input); setInput(''); }
    else if (e.key === 'ArrowUp') { e.preventDefault(); let idx = histIdx === -1 ? history.length-1 : Math.max(0, histIdx-1); setHistIdx(idx); setInput(history[idx] || ''); }
    else if (e.key === 'ArrowDown') { e.preventDefault(); let idx = histIdx === -1 ? -1 : Math.min(history.length-1, histIdx+1); setHistIdx(idx); setInput(idx === -1 ? '' : history[idx] || ''); }
  };

  return React.createElement('div', { className: 'h-full bg-[#0d1117] flex flex-col font-mono text-sm' },
    React.createElement('div', { ref: outRef, className: 'flex-1 overflow-auto p-3' },
      output.map((line, i) => React.createElement('div', { key: i, style: { color: line.color, fontSize: '12px', lineHeight: '1.6' } }, line.text))
    ),
    React.createElement('div', { className: 'flex items-center p-2 border-t border-[#2d333b]' },
      React.createElement('span', { className: 'text-[#2dd4bf] mr-2', style: { fontSize: '12px' } }, 'root@ares:~$'),
      React.createElement('input', { value: input, onChange: e => setInput(e.target.value), onKeyDown: handleKey, className: 'flex-1 bg-transparent outline-none text-[#e6edf3]', style: { fontSize: '12px', fontFamily: '"JetBrains Mono", monospace' }, spellCheck: false, autoFocus: true })
    )
  );
}

// ==================== DREV LAB APP ====================
function DrevLabApp() {
  const [projId, setProjId] = useState('155_HE_M107');
  const [chargeId, setChargeId] = useState('WB M4A2');
  const [range, setRange] = useState(10000);
  const [result, setResult] = useState(null);
  const [chartData, setChartData] = useState(null);
  const canvasRef = useRef(null);
  const chartRef = useRef(null);

  useEffect(() => {
    let proj = MUNITIONS_DB[projId];
    if (proj) setChargeId(Object.keys(proj.charges)[0]);
  }, [projId]);

  const calculate = () => {
    let proj = MUNITIONS_DB[projId];
    if (!proj) return;
    let p = { ...proj, v0: proj.charges[chargeId] || proj.v0 };
    let qe = PhysicsEngine.inverse(p, range);
    let sim = PhysicsEngine.simulate(p, qe);
    setResult({ qe: qe.toFixed(1), range: sim.range.toFixed(0), time: sim.time.toFixed(1), deflection: sim.deflection.toFixed(2), maxAlt: sim.maxAlt.toFixed(1) });
    setChartData(sim.points);
  };

  useEffect(() => {
    if (chartData && canvasRef.current) {
      if (chartRef.current) chartRef.current.destroy();
      chartRef.current = new Chart(canvasRef.current, {
        type: 'line',
        data: { labels: chartData.map(p => p.x), datasets: [{ label: 'Trayectoria', data: chartData.map(p => p.y), borderColor: '#2dd4bf', backgroundColor: 'rgba(45,212,191,0.1)', fill: true, tension: 0.3, pointRadius: 0 }] },

      });
    }
    return () => { if (chartRef.current) chartRef.current.destroy(); };
  }, [chartData]);

  return React.createElement('div', { className: 'h-full bg-[#0d1117] text-[#e6edf3] flex' },
    React.createElement('div', { className: 'w-72 p-4 border-r border-[#2d333b] space-y-3' },
      React.createElement('h3', { className: 'text-[#2dd4bf] font-rajdhani text-lg font-bold uppercase tracking-wider' }, 'DREV LAB'),
      React.createElement('label', { className: 'text-xs text-[#8b949e] block' }, 'Proyectil'),
      React.createElement('select', { value: projId, onChange: e => setProjId(e.target.value), className: 'w-full bg-[#161b22] border border-[#2d333b] rounded p-2 text-sm text-white' },
        Object.keys(MUNITIONS_DB).map(k => React.createElement('option', { key: k, value: k }, MUNITIONS_DB[k].name))
      ),
      React.createElement('label', { className: 'text-xs text-[#8b949e] block' }, 'Carga'),
      React.createElement('select', { value: chargeId, onChange: e => setChargeId(e.target.value), className: 'w-full bg-[#161b22] border border-[#2d333b] rounded p-2 text-sm text-white' },
        Object.keys(MUNITIONS_DB[projId]?.charges || {}).map(k => React.createElement('option', { key: k, value: k }, `${k} (${MUNITIONS_DB[projId].charges[k]} m/s)`))
      ),
      React.createElement('label', { className: 'text-xs text-[#8b949e] block' }, 'Alcance objetivo (m)'),
      React.createElement('input', { type: 'number', value: range, onChange: e => setRange(e.target.value), className: 'w-full bg-[#161b22] border border-[#2d333b] rounded p-2 text-sm text-white' }),

// ==================== WINDOW COMPONENT ====================
function WindowFrame({ win, children }) {
  const { activeWindowId, focusWindow, closeWindow, minimizeWindow, maximizeWindow } = useOSStore();
  const isActive = activeWindowId === win.id;
  const [drag, setDrag] = useState({ active: false, startX: 0, startY: 0, origX: 0, origY: 0 });

  const onMouseDown = (e) => {
    focusWindow(win.id);
    e.stopPropagation();
    if (e.target.closest('.window-header')) {
      setDrag({ active: true, startX: e.clientX, startY: e.clientY, origX: win.x, origY: win.y });
    }
  };

  useEffect(() => {
    const move = (e) => {
      if (!drag.active) return;
      const maxX = window.innerWidth - 200;
      const maxY = window.innerHeight - 56;
      useOSStore.setState(s => ({
        windows: s.windows.map(w => w.id === win.id ? { ...w, x: Math.max(0, Math.min(maxX, drag.origX + e.clientX - drag.startX)), y: Math.max(0, Math.min(maxY, drag.origY + e.clientY - drag.startY)) } : w)
      }));
    };
    const up = () => setDrag(prev => ({ ...prev, active: false }));
    if (drag.active) { window.addEventListener('mousemove', move); window.addEventListener('mouseup', up); }
    return () => { window.removeEventListener('mousemove', move); window.removeEventListener('mouseup', up); };
  }, [drag]);

  if (win.minimized) return null;

  const style = win.maximized
    ? { top: 0, left: 0, width: '100%', height: 'calc(100% - 48px)', zIndex: win.zIndex }
    : { top: win.y, left: win.x, width: win.width, height: win.height, zIndex: win.zIndex };

  return React.createElement('div', { className: 'window-frame', style, onMouseDown },
    React.createElement('div', { className: 'window-header' },
      React.createElement('span', { className: 'window-title' }, win.title),
      React.createElement('div', { className: 'window-controls' },
        React.createElement('button', { className: 'window-btn min', onClick: () => minimizeWindow(win.id) }),
        React.createElement('button', { className: 'window-btn max', onClick: () => maximizeWindow(win.id) }),
        React.createElement('button', { className: 'window-btn close', onClick: () => closeWindow(win.id) })
      )
    ),
    React.createElement('div', { className: 'window-body' }, children)
  );
}

          React.createElement('div', { className: 'text-[10px] font-bold text-[#64748b] uppercase px-2 py-1' }, cat),
          React.createElement('div', { className: 'start-app-grid' },
            apps.map(id => React.createElement('div', { key: id, className: 'start-app-item', onClick: () => { openWindow(id, { title: id.replace(/-/g,' ') }); toggleStartMenu(); } },
              React.createElement('span', { className: 'app-icon' }, APP_ICONS[id] || '📦'),
              React.createElement('span', null, id.replace(/-/g,' '))
            ))
          )
        )
      )
    ),

    // Taskbar
    React.createElement('div', { className: 'taskbar' },
      React.createElement('button', { onClick: toggleStartMenu, className: 'text-xl px-3 hover:bg-[#2dd4bf20] rounded-lg transition-colors', style:{background:showStartMenu?'rgba(45,212,191,.15)':''} }, '⬛'),
      React.createElement('div', { className: 'taskbar-center' },
        taskbarWindows.map(win => React.createElement('div', { key: win.id, className: `taskbar-icon ${activeWindowId===win.id?'active':''}`, onClick: () => focusWindow(win.id) }, APP_ICONS[win.appId] || '📦'))
      ),
      React.createElement('div', { className: 'taskbar-right' },
        React.createElement('span', null, '🌐 GW:192.168.88.1'),
        React.createElement('span', null, '🔒 VPN'),
        React.createElement('span', { className: 'font-mono' }, new Date().toLocaleTimeString('en-GB', { hour12: false }) + ' UTC')
      )
    )
  );
}

// ==================== LOGIN SCREEN ====================
function LoginScreen() {
  const { login } = useOSStore();
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [role, setRole] = useState('operator');
  const [host, setHost] = useState('localhost');
  const [bootLines, setBootLines] = useState([]);
  const [phase, setPhase] = useState('booting');

  useEffect(() => {
    const bootSequence = [
      { text: '[BOOT] ARES KERNEL v5.15.0-ares-sb #1 SMP', delay: 200 },
      { text: 'Loading essential drivers...', delay: 300 },
      { text: 'Mounting virtual filesystems...', delay: 400 },
      { text: 'Initializing Santa Barbara root partition...', delay: 500 },
      { text: 'Network interface eth0: 192.168.88.10/24 [OK]', delay: 600 },
      { text: 'DNS resolver: 1.1.1.1 [OK]', delay: 300 },
      { text: 'Proxmox VE cluster gateway: 192.168.88.1 [OK]', delay: 400 },
      { text: 'rclone mount verified [OK]', delay: 300 },
      { text: 'VPN tunnel established (WireGuard) [OK]', delay: 500 },
      { text: 'System ready. Welcome to ARES OS.', delay: 600 },
    ];
    let i = 0;
    const timer = setInterval(() => {
      if (i < bootSequence.length) { setBootLines(prev => [...prev, bootSequence[i].text]); i++; }
      else { clearInterval(timer); setTimeout(() => setPhase('login'), 500); }
    }, 200);
    return () => clearInterval(timer);
  }, []);

  const handleLogin = () => login({ username: username || 'Santa Barbara', role });

  return React.createElement('div', { className: 'login-container' },
    phase === 'booting' ? React.createElement('div', { className: 'login-terminal' },
      bootLines.map((line, i) => React.createElement('div', { key: i, style: { animation: 'fadeIn .3s' } }, line)),
      React.createElement('div', { className: 'mt-4 animate-pulse' }, '▊')
    ) : React.createElement('div', { className: 'login-panel' },
      React.createElement('div', { className: 'text-4xl mb-4' }, '🔴'),
      React.createElement('h1', { className: 'text-[#2dd4bf] font-rajdhani text-2xl font-bold uppercase tracking-widest mb-2' }, 'ARES OS'),
      React.createElement('p', { className: 'text-[#8b949e] text-sm mb-6' }, 'Sistema de Artillería de Campaña'),
      React.createElement('input', { value: username, onChange: e => setUsername(e.target.value), placeholder: 'Usuario (root: Santa Barbara)' }),
      React.createElement('input', { type: 'password', value: password, onChange: e => setPassword(e.target.value), placeholder: 'Contraseña', onKeyPress: e => e.key==='Enter'&&handleLogin() }),
      React.createElement('select', { value: role, onChange: e => setRole(e.target.value) },
        React.createElement('option', { value: 'root' }, 'Root'), React.createElement('option', { value: 'admin' }, 'Admin'),
        React.createElement('option', { value: 'operator' }, 'Operator'), React.createElement('option', { value: 'guest' }, 'Guest')
      ),
      React.createElement('select', { value: host, onChange: e => setHost(e.target.value) },
        React.createElement('option', { value: 'localhost' }, 'Servidor Local'), React.createElement('option', { value: 'remote' }, 'Servidor Remoto (192.168.88.1)')
      ),
      React.createElement('button', { onClick: handleLogin, className: 'login-btn' }, 'AUTENTICAR'),
      React.createElement('div', { className: 'mt-3 text-[10px] text-[#8b949e]' }, 'GW: 192.168.88.1 | DNS: 1.1.1.1 | VPN: WireGuard')
    )
  );
}

// ==================== MAIN APP ====================
function AresOS() {
  const { isAuthenticated } = useOSStore();
  return isAuthenticated ? React.createElement(Desktop) : React.createElement(LoginScreen);
}

// ==================== RENDER ====================
const root = createRoot(document.getElementById('root'));
root.render(React.createElement(AresOS));
