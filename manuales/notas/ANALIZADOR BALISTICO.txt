    def atmosphere_model(self, altitude):
        """Modelo atmosférico mejorado con humedad y corrección ICAO"""
        T0 = self.temp + 273.15
        P0 = self.pressure
        RH = self.humidity / 100.0
        
        # Presión de vapor saturado (Buck equation)
        e_sat = 611.21 * np.exp((18.678 - T0/234.5) * (T0/(257.14 + T0)))
        e = RH * e_sat
        P_dry = P0 - e
        
        # Perfil atmosférico según altitud
        if altitude <= 11000:  # Troposfera
            T = T0 - 0.0065 * altitude
            exponent = 5.256
        else:  # Estratosfera
            T = 216.65  # Isoterma
            exponent = 34.163
        
        # Cálculo de presión y densidad
        P = P0 * (T/T0)**exponent
        rho_dry = P_dry / (self.R * T)
        rho_vapor = e / (461.5 * T)  # Constante específica del vapor de agua
        rho = rho_dry + rho_vapor
        
        # Velocidad del sonido
        sonic = np.sqrt(self.gamma * self.R * T)
        
        return rho, sonic

    def drag_coefficient(self, mach):
        """Coeficiente de arrastre con interpolación spline y corrección BC"""
        mach = np.clip(mach, self.mach_points[0], self.mach_points[-1])
        return self.cd_spline(mach) * self.bc

    def wind_effect(self, altitude, vx, vy, vz):
        """Efecto del viento en la trayectoria"""
        # Perfil de viento (simplificado: constante con la altura)
        wind_x = self.wind_vel * np.cos(np.radians(self.wind_dir))
        wind_z = self.wind_vel * np.sin(np.radians(self.wind_dir))
        
        # Velocidad relativa al viento
        v_rel_x = vx - wind_x
        v_rel_z = vz - wind_z
        v_rel_mag = np.sqrt(v_rel_x**2 + vy**2 + v_rel_z**2)
        
        return v_rel_x, v_rel_z, v_rel_mag

    def ballistic_ode(self, t, state):
        """Ecuaciones diferenciales del movimiento balístico avanzado"""
        x, y, z, vx, vy, vz = state
        
        # Modelo atmosférico
        rho, sonic = self.atmosphere_model(y)
        
        # Efecto del viento
        v_rel_x, v_rel_z, v_rel_mag = self.wind_effect(y, vx, vy, vz)
        
        # Número de Mach
        mach = v_rel_mag / sonic if sonic > 0 else 0
        
        # Fuerza de arrastre
        drag_factor = 0.5 * rho * self.drag_coefficient(mach) * self.area / self.mass
        drag_x = -drag_factor * v_rel_mag * v_rel_x
        drag_y = -drag_factor * v_rel_mag * vy
        drag_z = -drag_factor * v_rel_mag * v_rel_z
        
        # Fuerza de gravedad
        gravity_y = -self.g
        
        # Efecto Coriolis
        lat_rad = np.radians(self.lat)
        coriolis_y = 2 * self.earth_omega * vz * np.sin(lat_rad)
        coriolis_z = -2 * self.earth_omega * vx * np.sin(lat_rad)
        
        return [
            vx,  # dx/dt
            vy,  # dy/dt
            vz,  # dz/dt
            drag_x,  # dvx/dt
            gravity_y + drag_y + coriolis_y,  # dvy/dt
            drag_z + coriolis_z  # dvz/dt
        ]

    def solve_trajectory(self, angulo_disparo, t_max=120, dt=0.1):
        """Resolver trayectoria con detección de impacto"""
        # Convertir ángulo a radianes
        theta_rad = np.radians(angulo_disparo)
        vx0 = self.v0 * np.cos(theta_rad)
        vy0 = self.v0 * np.sin(theta_rad)
        initial_state = [0, 0, 0, vx0, vy0, 0]
        
        # Evento de impacto en el suelo
        def hit_ground(t, state):
            return state[1]  # Altura y
        hit_ground.terminal = True
        hit_ground.direction = -1
        
        # Integración numérica
        solution = solve_ivp(
            fun=self.ballistic_ode,
            t_span=(0, t_max),
            y0=initial_state,
            events=[hit_ground],
            method='RK45',
            t_eval=np.arange(0, t_max, dt),
            rtol=1e-8,
            atol=1e-10
        )
        
        # Resultados de la trayectoria
        trajectory = {
            'time': solution.t,
            'x': solution.y[0],
            'y': solution.y[1],
            'z': solution.y[2],
            'vx': solution.y[3],
            'vy': solution.y[4],
            'vz': solution.y[5],
            'velocity': np.sqrt(solution.y[3]**2 + solution.y[4]**2 + solution.y[5]**2),
            'mach': np.sqrt(solution.y[3]**2 + solution.y[4]**2 + solution.y[5]**2) / 340.0,
            'status': solution.status,
            'message': solution.message,
            'impact_time': solution.t_events[0][0] if solution.t_events[0].size > 0 else t_max,
            'impact_range': solution.y_events[0][0][0] if solution.y_events[0].size > 0 else solution.y[0][-1]
        }
        return trajectory

class BalisticNeuralCore:
    """Aproximador ANN para dinámica de fluidos no lineal"""
    def __init__(self):
        self.W1 = np.random.randn(6, 12) * 0.1
        self.b1 = np.zeros(12)
        self.W2 = np.random.randn(12, 3) * 0.1
        self.b2 = np.zeros(3)

    def activar_sigmoide(self, x):
        return 1 / (1 + np.exp(-np.clip(x, -500, 500)))

    def activar_tanh(self, x):
        return np.tanh(x)

    def inferencia_aerodinamica(self, estado_vuelo):
        z1 = np.dot(estado_vuelo, self.W1) + self.b1
        a1 = self.activar_sigmoide(z1)
        z2 = np.dot(a1, self.W2) + self.b2
        return self.activar_tanh(z2)

class CNNTargetClassifier:
    """Coprocesador Convolucional: Clasificación de Escenas y Rastreo Táctico"""
    def __init__(self):
        self.kernel_relieve = np.array([[ 1,  1,  1],
                                        [ 0,  0,  0],
                                        [-1, -1, -1]])
        self.clases_terreno = [
            "🟩 Llanura Abierta", 
            "🏢 Zona Urbana / Estructuras", 
            "⛰️ Montaña Rocosa", 
            "🟦 Cuerpo de Agua"
        ]
        # Variables de rastreo de objetivo móvil
        self.target_pos = np.array([0.0, 0.0, 0.0])
        self.target_vel = np.array([0.0, 0.0, 0.0])

    def configurar_objetivo(self, punto_impacto_previsto, tiempo_vuelo):
        """Genera un objetivo móvil que cruzará cerca del punto de impacto"""
        # El objetivo empieza 1 km atrás y se mueve hacia el impacto
        distancia_inicial = 1000.0 
        self.target_pos = np.array([punto_impacto_previsto - distancia_inicial, 200.0, 0.0])
        # Velocidad necesaria para llegar al punto casi al mismo tiempo que el proyectil
        vel_x = distancia_inicial / tiempo_vuelo
        self.target_vel = np.array([vel_x, -200.0/tiempo_vuelo, 0.0])

    def actualizar_objetivo(self, dt):
        """Mueve el convoy enemigo en tierra"""
        self.target_pos += self.target_vel * dt
        return self.target_pos

    def escanear_escena(self, pos_proyectil):
        matriz_terreno = np.random.rand(10, 10) * np.sin(pos_proyectil[0] / 500)
        
        mapa_caracteristicas = np.zeros((8, 8))
        for i in range(8):
            for j in range(8):
                region = matriz_terreno[i:i+3, j:j+3]
                mapa_caracteristicas[i, j] = np.sum(region * self.kernel_relieve)
                
        mapa_relu = np.maximum(0, mapa_caracteristicas)
        activacion_maxima = np.max(mapa_relu)
        
        indice_clase = int((activacion_maxima * abs(pos_proyectil[2])) % 4)
        confianza = min(99.9, 50 + (activacion_maxima * 10))
        
        # Lógica de Alerta de Proximidad
        distancia_objetivo = np.linalg.norm(pos_proyectil - self.target_pos)
        alerta = ""
        if distancia_objetivo < 3000:
            alerta = "⚠️ OBJETIVO EN RÁDAR"
        if distancia_objetivo < 1000:
            alerta = "🚨 PELIGRO CERCANO - LOCK ON"
            
        return self.clases_terreno[indice_clase], confianza, alerta, distancia_objetivo

class CurvaEmpuje:
    def __init__(self):
        self.t_ignicion = 2.0
        self.duracion = 3.8
        self.empuje_max = 4500.0
        self.m_propelente = 6.8

    def empuje(self, t):
        if self.t_ignicion <= t <= (self.t_ignicion + self.duracion):
            return self.empuje_max
        return 0.0

    def masa_actual(self, t, m_inicial):
        if t < self.t_ignicion:
            return m_inicial
        elif t > self.t_ignicion + self.duracion:
            return m_inicial - self.m_propelente
        else:
            fraccion = (t - self.t_ignicion) / self.duracion
            return m_inicial - (fraccion * self.m_propelente)

class SantaBarbaraPhysicsEngine:
    def __init__(self, latitud_deg=40.0):
        self.lat_rad = np.radians(latitud_deg)
        self.omega_tierra = 7.292115e-5
        self.ann = BalisticNeuralCore()
        self.hera = CurvaEmpuje()

    def aceleracion_coriolis(self, vel):
        omega_vec = np.array([self.omega_tierra * np.cos(self.lat_rad), 0, -self.omega_tierra * np.sin(self.lat_rad)])
        return -2 * np.cross(omega_vec, vel)

    def _derivadas(self, t, S):
        pos, vel = S[0:3], S[3:6]
        v_mag = max(np.linalg.norm(vel), 1e-6)
        
        m_actual = self.hera.masa_actual(t, 43.5)
        thrust_mag = self.hera.empuje(t)
        f_thrust = thrust_mag * (vel / v_mag)

        estado_ann = np.array([-pos[2], v_mag, v_mag/340.0, m_actual, 0, 0])
        mods = self.ann.inferencia_aerodinamica(estado_ann)
        
        q_dyn = 0.5 * 1.225 * np.exp(pos[2]/8000) * v_mag**2
        area = np.pi * (0.155/2)**2
        
        f_drag = -(0.2 + 0.1 * mods[0]) * q_dyn * area * (vel / v_mag)
        f_lift = np.array([0, (0.05 * mods[1]) * q_dyn * area, 0]) 
        gravedad = np.array([0, 0, -9.80665]) * m_actual
        
        acc = ((f_drag + f_lift + f_thrust + gravedad) / m_actual) + self.aceleracion_coriolis(vel)
        return np.concatenate((vel, acc))

    def simular_mision(self, elevacion_deg, vel_boca):
        el_rad = np.radians(elevacion_deg)
        S0 = np.array([0, 0, 0, vel_boca * np.cos(el_rad), 0, vel_boca * np.sin(el_rad)])
        impacto = lambda t, y: y[2]
        impacto.terminal, impacto.direction = True, -1
        return solve_ivp(self._derivadas, [0, 200], S0, events=impacto, max_step=0.1, method='RK45')

# =============================================================================
# MOTOR BALÍSTICO 4-DoF - PROYECTO ARES CORE / OPEN CLAW
# ALGORITMOS CALIBRADOS SEGÚN MANUALES FM 6-40 / TM 43-0001-28 / FT 155-ADD-R-1

class EntornoBalistico:
    """Parametrización ambiental basada en doctrina de artillería."""
    def __init__(self):
        self.latitud = math.radians(14.08)  # Latitud operativa (Tegucigalpa, HN)
        self.acimut_fuego = math.radians(45.0)  # Dirección de tiro
        self.omega_tierra = 7.2921159e-5
        self.gravedad = 9.80665
        self.temp_aire_c = 15.0
        self.presion_hpa = 1013.25
        temp_k = self.temp_aire_c + 273.15
        self.rho_0 = (self.presion_hpa * 100) / (287.05 * temp_k)
        self.viento_x = -2.0  # Viento de frente
        self.viento_z = 1.5   # Viento cruzado

class CalculadoraMach:
    """Curva de arrastre G7 para proyectiles de artillería."""
    @staticmethod
    def obtener_cd_mach(mach: float, factor_forma: float) -> float:
        if mach < 0.8:
            cd_base = 0.12
        elif mach < 1.0:
            cd_base = 0.12 + 0.18 * ((mach - 0.8) / 0.2)
        elif mach < 1.2:
            cd_base = 0.30 - 0.05 * ((mach - 1.0) / 0.2)
        else:
            cd_base = 0.25 - 0.05 * ((mach - 1.2) / 2.0)
        return max(0.12, cd_base) * factor_forma

class ObusCore:
    def __init__(self, entorno: EntornoBalistico):
        self.env = entorno
        self.h_escala = 8500.0

    def simular(self, calibre, masa, v0, qe_mils, factor_forma, paso_estria, es_hera=False):
        dt = 0.01
        theta = (qe_mils * 2 * math.pi) / 6400.0
        
        x, y, z = 0.0, 0.0, 0.0
        vx = v0 * math.cos(theta)
        vy = v0 * math.sin(theta)
        vz = 0.0
        t = 0.0
        y_max = 0.0
        area = math.pi * (calibre / 2)**2

        while y >= 0:
            temp_y = (self.env.temp_aire_c + 273.15) - (0.0065 * y)
            rho = self.env.rho_0 * math.exp(-y / self.h_escala)
            velocidad_sonido = math.sqrt(1.4 * 287.05 * temp_y)
            
            v_rel_x = vx - self.env.viento_x
            v_rel_y = vy
            v_rel_z = vz - self.env.viento_z
            v_rel = math.sqrt(v_rel_x**2 + v_rel_y**2 + v_rel_z**2)
            mach = v_rel / velocidad_sonido
            
            # Arrastre Aerodinámico
            cd = CalculadoraMach.obtener_cd_mach(mach, factor_forma)
            f_drag = (rho * cd * area) / (2 * masa)
            
            # Efecto HERA (Asistencia por cohete: reduce resistencia y da empuje en ascenso)
            if es_hera and t < 15.0 and vy > 0:
                f_drag *= 0.65  # Simulación de empuje que compensa el arrastre
            
            # Coriolis
            cor_x = 2 * self.env.omega_tierra * vy * math.cos(self.env.latitud) * math.cos(self.env.acimut_fuego)
            cor_z = -2 * self.env.omega_tierra * (vx * math.sin(self.env.latitud) - vy * math.cos(self.env.latitud) * math.sin(self.env.acimut_fuego))
            
            # Spin Drift (Deriva por giro)
            spin_acel = (0.0125 * paso_estria * v_rel) / (v0 * 0.1)

            ax = -f_drag * v_rel * v_rel_x + cor_x
            ay = -self.env.gravedad - (f_drag * v_rel * v_rel_y)
            az = -f_drag * v_rel * v_rel_z + cor_z + spin_acel
            
            x += vx * dt
            y += vy * dt
            z += vz * dt
            vx += ax * dt
            vy += ay * dt
            vz += az * dt
            t += dt
            
            if y > y_max: y_max = y
            
        return x, z, t, y_max

// ------------------------------------------------------------

struct BallisticPoint {

    BallisticPoint(const std::string& cid, ChargeColor c, int z,
                   double mv, double rng, double el = 0.0, double fs = 0.0)
        : charge_id(cid), bag_color(c), zone(z),
          mv_mps(mv), max_range_m(rng), elev_mil(el), fuze_set_sec(fs) {}

// ------------------------------------------------------------
// Tabla balística por sistema de armas

struct BallisticTable {
    std::string weapon_system;  // "M198/M199", "M114A2", etc.

    // Interpolación lineal de MV dada la zona
    double interpolate_mv(int zone_low, int zone_high, double fraction) const {
        const BallisticPoint* low  = nullptr;
        const BallisticPoint* high = nullptr;
        for (const auto& p : points) {
            if (p.zone == zone_low)  low  = &p;
            if (p.zone == zone_high) high = &p;
        }
        if (!low || !high) throw std::out_of_range("Zone not found in table");
        return low->mv_mps + fraction * (high->mv_mps - low->mv_mps);
    }

    // Obtener punto exacto por zona y carga
    const BallisticPoint* get_point(int zone, ChargeColor color) const {
        for (const auto& p : points) {
            if (p.zone == zone && p.bag_color == color) return &p;
        }
        return nullptr;
    }
};

// ------------------------------------------------------------
// Proyectil base
// ------------------------------------------------------------

struct Projectile {
    std::string   designation;
    Caliber       caliber;
    ProjectileType type;
    double        weight_lb;
    std::string   source_tm;

    // Cargas propulsoras compatibles (IDs)
    std::vector<std::string> compatible_charges;
    // Cargas prohibidas con razón
    std::unordered_map<std::string,std::string> prohibited_charges;

    // Restricciones operacionales
    std::vector<std::string> restrictions;

    // Tablas balísticas por sistema de armas
    std::unordered_map<std::string, BallisticTable> ballistic_tables;

    bool is_charge_compatible(const std::string& charge_id) const {
        if (prohibited_charges.count(charge_id)) return false;
        for (const auto& c : compatible_charges)
            if (c == charge_id) return true;
        return false;
    }

    std::string get_prohibition_reason(const std::string& charge_id) const {
        auto it = prohibited_charges.find(charge_id);
        if (it != prohibited_charges.end()) return it->second;
        return "";
    }
};

// ============================================================

class AmmunitionRegistry {
public:
    AmmunitionRegistry() {
        // 105mm
        add(make_M1_HE());
        add(make_M84_HC_SMOKE());
        add(make_M314A3_ILLUM());
        // 155mm
        add(make_M107_HE());
        add(make_M549A1_HERA());
        add(make_M718A1_AT());
        add(make_M485_ILLUM());
    }

    const Projectile& get(const std::string& designation) const {
        auto it = db_.find(designation);
        if (it == db_.end())
            throw std::invalid_argument("Projectile not found: " + designation);
        return it->second;
    }

    bool validate_combination(const std::string& proj, const std::string& charge,
                             # En generar_tabla_155mm() o función equivalente:
                            # 1. Asegurar que el nuevo proyectil aparece en PROY_155
                            # 2. Asegurar que sus cargas aparecen en CARGAS_155["proyectiles"]
                            # 3. Ejecutar generador y validar salida FT_155MM_CALIBRADA.txt
                            
                            tabla = generar_tabla_155mm()
                            with open("FT_155MM_CALIBRADA.txt", "w", encoding="utf-8") as f:
                                f.write(tabla)                              std::string& reason) const {
        const auto& p = get(proj);
        if (!p.is_charge_compatible(charge)) {
            reason = p.get_prohibition_reason(charge);
            if (reason.empty()) reason = "Carga no listada como compatible";
            return false;
        }
        return true;
    }

private:
    void add(Projectile p) { db_[p.designation] = std::move(p); }
    std::unordered_map<std::string, Projectile> db_;
};
