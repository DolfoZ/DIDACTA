class PhysicsEngine:
    G0 = 9.80665
    R_AIR = 287.05
    GAMMA = 1.4
    OMEGA = 7.2921159e-5
    RADIO_T = 6371000.0
    LAT = math.radians(14.08)  # Honduras (operativa)

    G7_TABLE = [
        (0.00, 0.120), (0.70, 0.122), (0.85, 0.135), (0.95, 0.200),
        (1.00, 0.350), (1.05, 0.380), (1.10, 0.375), (1.20, 0.355),

    ]

    PROYECTILES = {
        "155_HE_M107": {"nombre": "155mm HE M107", "calibre_mm": 155.0, "masa_kg": 43.2, "v0_ref": 563.0, "bc_g7": 0.185, "ff": 1.000, "cargas": ["GB_M3A1", "WB_M4A2", "WB_M119A2", "RB_M203"], "fuente": "FT 155-AM-2", "es_hera": False},
        "155_HERA_M549A1": {"nombre": "155mm HERA M549A1", "calibre_mm": 155.0, "masa_kg": 43.5, "v0_ref": 600.0, "bc_g7": 0.210, "ff": 0.950, "cargas": ["WB_M4A2", "WB_M119A2", "RB_M203"], "fuente": "TM 43-0001-28", "es_hera": True, "empuje_n": 1800.0, "duracion_hera_s": 0.8},
        "155_ILLUM_M485": {"nombre": "155mm ILLUM M485", "calibre_mm": 155.0, "masa_kg": 45.4, "v0_ref": 630.0, "bc_g7": 0.165, "ff": 1.020, "cargas": ["WB_M4A2", "WB_M119A2"], "fuente": "FM 6-40", "es_hera": False, "tia_ejection": True},
        "105_HE_M1": {"nombre": "105mm HE M1", "calibre_mm": 105.0, "masa_kg": 14.97, "v0_ref": 494.0, "bc_g7": 0.150, "ff": 1.042, "cargas": ["105_CH7"], "fuente": "FT 105-H-6", "es_hera": False}
    }
    CARGAS = {
        "105_CH7": {"designacion": "Charge 7 (M67)", "zona": 7, "v0_mps": 472.0, "alcance_max_km": 11.27},
        "GB_M3A1": {"designacion": "GB M3A1 (Z1-3)", "zona": 3, "v0_mps": 375.0, "alcance_max_km": 6.0},
        "WB_M4A2": {"designacion": "WB M4A2 (Z4-6)", "zona": 6, "v0_mps": 565.0, "alcance_max_km": 14.6},
        "WB_M119A2": {"designacion": "WB M119A2 (Z7-8)", "zona": 8, "v0_mps": 685.0, "alcance_max_km": 18.1},
        "RB_M203": {"designacion": "RB M203 (Z9)", "zona": 9, "v0_mps": 827.0, "alcance_max_km": 30.0}
    }

    @classmethod
    def interpolar_g7(cls, mach: float) -> float:
        t = cls.G7_TABLE
        if mach <= t[0][0]: return t[0][1]
        if mach >= t[-1][0]: return t[-1][1]
        for i in range(len(t)-1):
            if t[i][0] <= mach <= t[i+1][0]:
                m1,c1,m2,c2 = t[i][0],t[i][1],t[i+1][0],t[i+1][1]
                return c1 + (c2-c1)*(mach-m1)/(m2-m1)
        return 0.25

    @classmethod
    def atmosfera(cls, h: float) -> Tuple[float, float]:
        if h <= 11000:
            T = 288.15 - 0.0065*h
            P = 101325.0 * (T/288.15) ** (-9.80665/(0.0065*cls.R_AIR))
        else:
            T = 216.65
            P_trop = 22632.1
            P = P_trop * math.exp(-cls.G0*(h-11000)/(cls.R_AIR*T))
        return P/(cls.R_AIR*T), T

    @classmethod
    def gravedad(cls, h: float) -> float:
        gl = cls.G0 * (1 + 0.0053024*math.sin(cls.LAT)**2 - 0.0000058*math.sin(2*cls.LAT)**2)
        return gl * (cls.RADIO_T/(cls.RADIO_T+h))**2

    @classmethod
    def coriolis(cls, vx, vy, vz):
        om, lat = cls.OMEGA, cls.LAT
        return 2*om*(vy*math.sin(lat)-vz*math.cos(lat)), -2*om*vx*math.sin(lat), 2*om*vx*math.cos(lat)

    @classmethod
    def spin_drift(cls, rango_m: float, calibre_mm: float) -> float:
        k = 1.8e-5 if calibre_mm > 130 else 2.5e-5
        f = 1.0 + (0.02 if calibre_mm > 130 else 0.015)*(rango_m/1000.0)
        return k*(rango_m/1000.0)**2*f*318.31  # mils

    @classmethod
    def integrar_rk4(cls, proy: Dict, qe_mils: float, vz_w: float = 0.0, dt: float = 0.05) -> Dict:
        theta = qe_mils * 0.001
        v0 = proy["v0_actual"]
        x, y, z = 0.0, 0.5, 0.0
        vx = v0 * math.cos(theta)
        vy = v0 * math.sin(theta)
        vz = 0.0
        t = 0.0
        max_z = 0.0
        area = math.pi * (proy["calibre_mm"]/2000.0)**2
        masa = proy["masa_kg"]

        # HERA: empuje variable
        es_hera = proy.get("es_hera", False)
        empuje_n = proy.get("empuje_n", 0.0)
        duracion_hera = proy.get("duracion_hera_s", 0.0)

        while y >= 0 and t < 300:
            rho, temp = cls.atmosfera(y)
            a_sonido = math.sqrt(cls.GAMMA * cls.R_AIR * temp)
            vx_rel = vx
            vy_rel = vy
            vz_rel = vz - vz_w
            v_rel = math.sqrt(vx_rel**2 + vy_rel**2 + vz_rel**2) or 1e-6
            mach = v_rel / a_sonido
            cd = cls.interpolar_g7(mach) * proy["ff"]
            drag = 0.5 * rho * cd * area / masa
            cx, cy, cz = cls.coriolis(vx, vy, vz)
            g = cls.gravedad(y)

            # Empuje (solo si HERA activo)
            empuje = 0.0
            if es_hera and t < duracion_hera:
                empuje = empuje_n / masa

            ax = -drag * v_rel * vx_rel + cx
            ay = -drag * v_rel * vy_rel - g + cy + (empuje * (vy_rel / v_rel) if v_rel > 0 else 0)
            az = -drag * v_rel * vz_rel + cz

            vx += ax * dt
            vy += ay * dt
            vz += az * dt
            x += vx * dt
            y += vy * dt
            z += vz * dt
            t += dt
            if y > max_z: max_z = y

        # Interpolación de impacto
        if y < 0:
            frac = -y / vy if vy != 0 else 0
            x += vx * frac
            z += vz * frac
            t -= dt - frac

        return {"alcance_m": x, "tiempo_s": t, "max_alt_m": max_z, "deriva_z_m": z}

    @classmethod
    def inverso(cls, proy: Dict, rango_obj: float, tol: float = 1.0) -> float:
        lo, hi = 50.0, 1600.0
        for _ in range(60):
            mid = (lo+hi)/2.0
            sim = cls.integrar_rk4(proy, mid)
            if abs(sim["alcance_m"] - rango_obj) < tol: return mid
            if sim["alcance_m"] < rango_obj: lo = mid
            else: hi = mid
        return (lo+hi)/2.0

GHIDRA_SCRIPT = """// Ghidra Script: extract_ballistics.java
// @category: Analysis
// @description: Extrae tablas de constantes flotantes de un binario

import ghidra.app.script.GhidraScript;
import ghidra.program.model.listing.*;
import ghidra.program.model.data.*;
import java.io.*;
import java.util.*;
import java.util.regex.*;

public class extract_ballistics extends GhidraScript {
    @Override
    protected void run() throws Exception {
        File outFile = new File(currentProgram.getExecutablePath() + ".json");
        JSONObject result = new JSONObject();
        JSONArray tables = new JSONArray();
        
        DataIterator dataIter = currentProgram.getListing().getDefinedData(true);
        while (dataIter.hasNext()) {
            Data data = dataIter.next();
            if (data.isArray() && (data.getDataType().getName().contains("float") || data.getDataType().getName().contains("double"))) {
                List<Double> vals = new ArrayList<>();
                for (Data comp : data.getComponents()) {
                    Object val = comp.getValue();
                    if (val instanceof Number) {
                        vals.add(((Number)val).doubleValue());
                    } else if (val instanceof String) {
                        try { vals.add(Double.parseDouble((String)val)); } catch(Exception e) {}
                    }
                }
                if (vals.size() > 0) {
                    JSONObject arr = new JSONObject();
                    arr.put("address", data.getAddress().toString());
                    arr.put("type", data.getDataType().getName());
                    arr.put("values", vals);
                    tables.put(arr);
                }
            }
        }
        result.put("tables", tables);
        try (FileWriter fw = new FileWriter(outFile)) { fw.write(result.toString(2)); }
        println("Extracted " + tables.length() + " constant tables to " + outFile.getAbsolutePath());
    }
}
"""

def ghidra_extract_binary(binary_path: str) -> Dict:
    """Ejecuta Ghidra headless en el binario y devuelve las tablas extraídas."""
    if not os.path.exists(GHIDRA_INSTALL_DIR):
        return {"error": f"Ghidra no encontrado en {GHIDRA_INSTALL_DIR}. Instale Ghidra y configure GHIDRA_INSTALL_DIR."}
    with tempfile.TemporaryDirectory() as proj_dir:
        script_file = os.path.join(proj_dir, "extract_ballistics.java")
        with open(script_file, "w") as f:
            f.write(GHIDRA_SCRIPT)
        cmd = [
            os.path.join(GHIDRA_INSTALL_DIR, "support", "analyzeHeadless"),
            proj_dir, "tmp_proj",
            "-import", binary_path,
            "-postScript", script_file
        ]
        result = subprocess.run(cmd, capture_output=True, text=True)
        if result.returncode != 0:
            return {"error": result.stderr}
        output_json = binary_path + ".json"
        if os.path.exists(output_json):
            with open(output_json, "r") as f:
                data = json.load(f)
            os.unlink(output_json)
            return data
        return {"error": "No se generó archivo de salida"}

def calibrar_proyectil(proy_id: str, carga_id: str, csv_data: bytes) -> Dict:
    """Ajusta factor de forma y parámetros HERA usando un CSV."""
    import io
    df = pd.read_csv(io.BytesIO(csv_data))
    if "Range(m)" not in df.columns or "QE(mils)" not in df.columns:
        return {"error": "CSV debe contener columnas 'Range(m)' y 'QE(mils)'"}
    
    proy = PhysicsEngine.PROYECTILES[proy_id].copy()
    carga = PhysicsEngine.CARGAS[carga_id]
    proy["v0_actual"] = carga["v0_mps"]
    
    es_hera = proy.get("es_hera", False)
    
    def cost(params):
        if es_hera:
            ff, empuje, dura = params
            proy["ff"] = ff
            proy["empuje_n"] = empuje
            proy["duracion_hera_s"] = dura
        else:
            ff = params[0]
            proy["ff"] = ff
        err = 0.0
        for _, row in df.iterrows():
            sim = PhysicsEngine.integrar_rk4(proy, row["QE(mils)"])
            err += ((sim["alcance_m"] - row["Range(m)"]) / row["Range(m)"])**2
        return err / len(df)
    
    if es_hera:
        bounds = [(0.8,1.2), (500,5000), (0.3,2.5)]
        res = differential_evolution(cost, bounds, maxiter=30, popsize=10, seed=42)
        ff, emp, dur = res.x
        PhysicsEngine.PROYECTILES[proy_id]["ff"] = ff
        PhysicsEngine.PROYECTILES[proy_id]["empuje_n"] = emp
        PhysicsEngine.PROYECTILES[proy_id]["duracion_hera_s"] = dur
        return {"status": "ok", "ff": ff, "empuje_n": emp, "duracion_s": dur, "error": res.fun}
    else:
        bounds = [(0.9,1.1)]
        res = differential_evolution(cost, bounds)
        ff = res.x[0]
        PhysicsEngine.PROYECTILES[proy_id]["ff"] = ff
        return {"status": "ok", "ff": ff, "error": res.fun}

@app.post("/api/solve")
async def solve(proyectil_id: str = Form(...), charge_id: str = Form(...), range_m: float = Form(...)):
    proy = PhysicsEngine.PROYECTILES[proyectil_id].copy()
    carga = PhysicsEngine.CARGAS[charge_id]
    proy["v0_actual"] = carga["v0_mps"]
    qe = PhysicsEngine.inverso(proy, range_m)
    sim = PhysicsEngine.integrar_rk4(proy, qe)
    drift = PhysicsEngine.spin_drift(sim["alcance_m"], proy["calibre_mm"])
    return {"qe_mils": qe, **sim, "drift_mils": drift}

@app.post("/api/table")
async def generate_table(proyectil_id: str = Form(...), charge_id: str = Form(...), step_km: float = 2.0):
    proy = PhysicsEngine.PROYECTILES[proyectil_id].copy()
    carga = PhysicsEngine.CARGAS[charge_id]
    proy["v0_actual"] = carga["v0_mps"]
    max_range = carga["alcance_max_km"] * 1000
    ranges = list(range(1000, int(max_range), int(step_km*1000)))
    tabla = []
    for r in ranges:
        qe = PhysicsEngine.inverso(proy, r)
        sim = PhysicsEngine.integrar_rk4(proy, qe)
        drift = PhysicsEngine.spin_drift(sim["alcance_m"], proy["calibre_mm"])
        tabla.append({"Rango_m": r, "QE_mils": round(qe,1), "TOF_s": round(sim["tiempo_s"],2), "Drift_mils": round(drift,2)})
    return tabla

@app.post("/api/ghidra/analyze")
async def ghidra_analyze(file: UploadFile = File(...)):
    tmp = Path(f"/tmp/{file.filename}")
    with open(tmp, "wb") as f:
        f.write(await file.read())
    result = ghidra_extract_binary(str(tmp))
    tmp.unlink()
    return result

@app.post("/api/calibrate")
async def calibrate(proyectil_id: str = Form(...), charge_id: str = Form(...), csv_file: UploadFile = File(...)):
    csv_data = await csv_file.read()
    return calibrar_proyectil(proyectil_id, charge_id, csv_data)
