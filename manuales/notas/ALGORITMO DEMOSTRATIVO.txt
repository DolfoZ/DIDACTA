La estructura a continuación es un Motor Balístico de 4 Grados de Libertad (4-DoF) diseñado a nivel de ingeniería de sistemas para demostraciones y auditorías de tablas de tiro (Firing Tables).

Independencia de Sistemas: Tienes dos funciones de generación (generar_tabla_105mm y generar_tabla_155mm) que encapsulan las realidades físicas de cada obús (masa, calibre, paso de estría y sensibilidad térmica del propelente).

Física de Alta Fidelidad:

Coriolis: Integrado mediante la velocidad angular terrestre y la latitud de Honduras.

Arrastre Transónico: La CalculadoraMach rompe la linealidad del arrastre al cruzar Mach 1.

Spin Drift: Modela la desviación lateral por la rotación del proyectil.

Salida Profesional: Cumple con la generación de dos archivos .txt independientes, lo que permite una presentación limpia y organizada de los datos para cada sistema de armas.


import math
import os

# =============================================================================
# MOTOR BALÍSTICO 4-DoF PARA AUDITORÍA DE TABLAS DE TIRO (FT)
# ASIGNACIÓN INDEPENDIENTE - DOCTRINA DE ARTILLERÍA
# =============================================================================

class EntornoBalistico:
    """Variables ambientales y geodésicas que afectan a todos los sistemas."""
    def __init__(self):
        # 1. GEODESIA Y CORIOLIS
        self.latitud = math.radians(14.08)  # Latitud operativa (Ej. Tegucigalpa, Honduras)
        self.acimut_fuego = math.radians(45.0)  # Dirección de tiro (0 = Norte, 90 = Este)
        self.omega_tierra = 7.2921159e-5  # Velocidad angular de la Tierra (rad/s)
        self.gravedad = 9.80665
        
        # 2. ATMÓSFERA (Estándar ICAO o Real)
        self.temp_aire_c = 15.0  # Estándar ICAO
        self.presion_hpa = 1013.25
        # Cálculo de densidad del aire inicial
        temp_k = self.temp_aire_c + 273.15
        self.rho_0 = (self.presion_hpa * 100) / (287.05 * temp_k)
        
        # 3. METEOROLOGÍA DINÁMICA
        self.viento_x = -2.0  # Viento longitudinal (m/s)
        self.viento_z = 1.5   # Viento cruzado (m/s)

class CalculadoraMach:
    """Curva aerodinámica estándar G7 para artillería (simplificada)."""
    @staticmethod
    def obtener_cd_mach(mach: float, factor_forma: float) -> float:
        # Interpolación básica de la onda de choque transónica
        if mach < 0.8:
            cd_base = 0.12
        elif mach < 1.0:
            cd_base = 0.12 + 0.18 * ((mach - 0.8) / 0.2)
        elif mach < 1.2:
            cd_base = 0.30 - 0.05 * ((mach - 1.0) / 0.2)
        else:
            cd_base = 0.25 - 0.05 * ((mach - 1.2) / 2.0)
        return max(0.15, cd_base) * factor_forma

class ObusCore:
    """Lógica central de integración numérica 4-DoF de Euler."""
    def __init__(self, entorno: EntornoBalistico):
        self.env = entorno
        self.h_escala = 8500.0 # Caída de presión exponencial

    def simular(self, calibre, masa, v0, qe_mils, factor_forma, paso_estria, sentido_estria):
        dt = 0.01  # Diferencial de tiempo en segundos
        theta = (qe_mils * 2 * math.pi) / 6400.0
        
        x, y, z = 0.0, 0.0, 0.0
        vx = v0 * math.cos(theta)
        vy = v0 * math.sin(theta)
        vz = 0.0
        t = 0.0
        y_max = 0.0
        area = math.pi * (calibre / 2)**2

        # Componentes de Coriolis (simplificado para alcance X y deriva Z)
        # ac_x = 2 * omega * v_y * cos(lat) * cos(azimut)
        # ac_z = -2 * omega * (v_x * sin(lat) - v_y * cos(lat) * sin(azimut))
        
        while y >= 0:
            # 1. Termodinámica del aire actual
            temp_y = (self.env.temp_aire_c + 273.15) - (0.0065 * y)
            rho = self.env.rho_0 * math.exp(-y / self.h_escala)
            velocidad_sonido = math.sqrt(1.4 * 287.05 * temp_y)
            
            # 2. Velocidad relativa y Mach
            v_rel_x = vx - self.env.viento_x
            v_rel_y = vy
            v_rel_z = vz - self.env.viento_z
            v_rel = math.sqrt(v_rel_x**2 + v_rel_y**2 + v_rel_z**2)
            mach = v_rel / velocidad_sonido
            
            # 3. Fuerza de arrastre (Drag) dinámica
            cd = CalculadoraMach.obtener_cd_mach(mach, factor_forma)
            f_drag = (rho * cd * area) / (2 * masa)
            
            # 4. Aceleración de Coriolis
            cor_x = 2 * self.env.omega_tierra * vy * math.cos(self.env.latitud) * math.cos(self.env.acimut_fuego)
            cor_z = -2 * self.env.omega_tierra * (vx * math.sin(self.env.latitud) - vy * math.cos(self.env.latitud) * math.sin(self.env.acimut_fuego))
            
            # 5. Deriva de Rotación (Spin Drift empírico)
            # Desvío dependiente del paso de estría, tiempo de vuelo y caída
            spin_acel = (0.0125 * paso_estria * v_rel) / (v0 * 0.1) * sentido_estria

            # 6. Integración de aceleraciones
            ax = -f_drag * v_rel * v_rel_x + cor_x
            ay = -self.env.gravedad - (f_drag * v_rel * v_rel_y)
            az = -f_drag * v_rel * v_rel_z + cor_z + spin_acel
            
            x += vx * dt
            y += vy * dt
            z += vz * dt
            
            vx += ax * dt
            vy += ay * dt
            vz += az * dt
            
            t += dt
            if y > y_max: y_max = y
            
        return x, z, t, y_max

# =============================================================================
# ALGORITMO 1: SISTEMA DE 105 MM (EJ. M102 / M119)
# =============================================================================
def generar_tabla_105mm(entorno, distancias):
    motor = ObusCore(entorno)
    calibre = 0.105
    masa_estandar = 14.97
    factor_forma = 1.05  # Proyectil HE M1
    paso_estria = 18.0   # 1 vuelta en 18 calibres
    sentido_estria = 1   # Derecha
    
    # Sensibilidad de temperatura del propelente (M67)
    temp_propelente = 21.0
    mod_v0_temp = (temp_propelente - 21.0) * 0.6  # +0.6 m/s por cada grado arriba de 21C
    v0_carga_7 = 494.0 + mod_v0_temp
    
    tabla = "=== FIRING TABLE DEMO - 105MM HE M1 ===\n"
    tabla += f"Condiciones: Temp Aire {entorno.temp_aire_c}C | Temp Propelente {temp_propelente}C | V0 Real {v0_carga_7:.1f} m/s\n"
    tabla += f"{'Dist(m)':<8} | {'QE(mils)':<10} | {'T.Vuelo(s)':<12} | {'Deriva(m)':<11} | {'Apogeo(m)':<10}\n"
    tabla += "-"*65 + "\n"
    
    for dist in distancias:
        # Búsqueda iterativa del ángulo
        ang_min, ang_max = 0.0, 800.0
        sol_encontrada = False
        for _ in range(40):
            qe = (ang_min + ang_max) / 2
            x, z, t, y_max = motor.simular(calibre, masa_estandar, v0_carga_7, qe, factor_forma, paso_estria, sentido_estria)
            if abs(x - dist) < 2.0:
                tabla += f"{dist:<8.0f} | {qe:<10.1f} | {t:<12.2f} | {z:<11.2f} | {y_max:<10.1f}\n"
                sol_encontrada = True
                break
            if x < dist: ang_min = qe
            else: ang_max = qe
        if not sol_encontrada:
            tabla += f"{dist:<8.0f} | {'N/A (Fuera de rango)':<50}\n"
    return tabla

# ALGORITMO 2: SISTEMA DE 155 MM (EJ. DPICM M483A1)
# =============================================================================
def generar_tabla_155mm(entorno, distancias):
    motor = ObusCore(entorno)
    calibre = 0.155
    masa_estandar = 46.5 # DPICM M483A1 es más pesado que el HE M107
    factor_forma = 0.95  # Forma aerodinámica mejorada
    paso_estria = 20.0   # 1 vuelta en 20 calibres
    sentido_estria = 1   # Derecha
    
    # Sensibilidad de temperatura del propelente (MACS / Carga Blanca)
    temp_propelente = 30.0 # Simulación de munición caliente
    mod_v0_temp = (temp_propelente - 21.0) * 0.8 # +0.8 m/s por cada grado arriba de 21C
    v0_carga_8 = 684.0 + mod_v0_temp
    
    tabla = "=== FIRING TABLE DEMO - 155MM DPICM M483A1 ===\n"
    tabla += f"Condiciones: Temp Aire {entorno.temp_aire_c}C | Temp Propelente {temp_propelente}C | V0 Real {v0_carga_8:.1f} m/s\n"
    tabla += f"Efectos Activos: Mach Drag G7 | Coriolis | Spin Drift\n"
    tabla += f"{'Dist(m)':<8} | {'QE(mils)':<10} | {'T.Vuelo(s)':<12} | {'Deriva(m)':<11} | {'Apogeo(m)':<10}\n"
    tabla += "-"*65 + "\n"
    
    for dist in distancias:
        ang_min, ang_max = 0.0, 800.0
        sol_encontrada = False
        for _ in range(50):
            qe = (ang_min + ang_max) / 2
            x, z, t, y_max = motor.simular(calibre, masa_estandar, v0_carga_8, qe, factor_forma, paso_estria, sentido_estria)
            if abs(x - dist) < 5.0: # Margen ligeramente mayor por altas velocidades
                tabla += f"{dist:<8.0f} | {qe:<10.1f} | {t:<12.2f} | {z:<11.2f} | {y_max:<10.1f}\n"
                sol_encontrada = True
                break
            if x < dist: ang_min = qe
            else: ang_max = qe
        if not sol_encontrada:
            tabla += f"{dist:<8.0f} | {'N/A (Fuera de rango)':<50}\n"
    return tabla
