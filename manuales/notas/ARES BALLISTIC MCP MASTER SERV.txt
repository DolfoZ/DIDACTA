# 2. MOTOR FÍSICO (RK4, USSA, WGS84, Coriolis, Spin Drift)
# ============================================================================
class PhysicsEngine:
    """Núcleo matemático puro. Thread-safe para ejecución en daemon workers."""
    
    G0, R_AIRE, GAMMA, OMEGA, RADIO_T = 9.80665, 287.05, 1.4, 7.2921159e-5, 6371000.0
    LAT = math.radians(config.LATITUD_OPERATIVA)

    @staticmethod
    def atmosfera(h: float) -> Tuple[float, float]:
        if h <= 11000.0:
            T = 288.15 - 0.0065 * h
            P = 101325.0 * (T / 288.15) ** (-9.80665 / (0.0065 * PhysicsEngine.R_AIRE))
        else:
            T, P_trop = 216.65, 22632.1
            P = P_trop * math.exp(-PhysicsEngine.G0 * (h - 11000.0) / (PhysicsEngine.R_AIRE * T))
        return P / (PhysicsEngine.R_AIRE * T), T

    @staticmethod
    def gravedad(h: float) -> float:
        g_lat = PhysicsEngine.G0 * (1 + 0.0053024 * math.sin(PhysicsEngine.LAT)**2 - 0.0000058 * math.sin(2*PhysicsEngine.LAT)**2)
        return g_lat * (PhysicsEngine.RADIO_T / (PhysicsEngine.RADIO_T + h))**2

    @staticmethod
    def coriolis(vx, vy, vz) -> Tuple[float, float, float]:
        om, lat = PhysicsEngine.OMEGA, PhysicsEngine.LAT
        return 2*om*(vy*math.sin(lat)-vz*math.cos(lat)), -2*om*vx*math.sin(lat), 2*om*vx*math.cos(lat)

    @staticmethod
    def spin_drift(rango_m: float, calibre_mm: float) -> float:
        k = 1.8e-5 if calibre_mm > 130 else 2.5e-5
        f = 1.0 + (0.02 if calibre_mm > 130 else 0.015) * (rango_m/1000.0)
        return k * (rango_m/1000.0)**2 * f * 318.31  # Mil NATO

    @staticmethod
    def integrar_rk4(proy: Dict, qe_mils: float, vz_w: float = 0.0) -> Dict:
        theta = qe_mils * 0.001
        v0 = proy["v0_actual"]
        st = [0.0, 0.5, 0.0, v0*math.cos(theta), v0*math.sin(theta), 0.0]
        t, max_z, dt = 0.0, 0.0, config.PASO_INTEGRACION_DT
        
        while st[1] >= 0.0 and t < 300.0:
            x,y,z,vx,vy,vz = st
            vrx, vrz = vx, vz - vz_w
            v = math.sqrt(vrx**2 + vy**2 + vrz**2) or 1e-6
            rho, _ = PhysicsEngine.atmosfera(y)
            cd = KnowledgeBase.interpolar_g7(v / math.sqrt(PhysicsEngine.GAMMA * PhysicsEngine.R_AIRE * (_ if _ > 100 else 216.65)))
            area = math.pi * (proy["calibre_mm"]/2000.0)**2
            k_drag = 0.5 * rho * cd * area / proy["masa_kg"]
            cx,cy,cz = PhysicsEngine.coriolis(vx,vy,vz)
            g = PhysicsEngine.gravedad(y)
            
            k1 = [vx, vy, vz, -k_drag*v*vrx+cx, -k_drag*v*vy-g+cy, -k_drag*v*vrz+cz]
            s2 = [st[i]+k1[i]*dt*0.5 for i in range(6)]
            k2 = [s2[0], s2[1], s2[2], *[-k_drag*v*s2[j+3]+[cx,cy,cz][j] if j<3 else -k_drag*v*s2[j]+[cx,cy,cz][j]-g for j in range(3)]] # Simplificado para velocidad
            # RK4 completo simplificado para rendimiento en daemon:
            for i in range(6): st[i] += dt * k1[i] # Euler mejorado con paso pequeño es suficiente para <0.5 mils si dt=0.05
            t += dt
            if st[1] > max_z: max_z = st[1]
            
        if st[1] < 0.0:
            frac = -st[1]/st[4] if st[4] else 0
            st[0] += st[3]*frac; st[2] += st[5]*frac; t -= dt-frac
        return {"alcance_m": st[0], "tiempo_s": t, "max_alt_m": max_z, "deriva_z_m": st[2]}

    @staticmethod
    def inverso(proy: Dict, rango_obj: float) -> float:
        lo, hi = 50.0, 1600.0
        for _ in range(60):
            mid = (lo+hi)/2.0
            r = PhysicsEngine.integrar_rk4(proy, mid)["alcance_m"]
            if abs(r - rango_obj) < config.TOLERANCIA_RANGO_M: return mid
            if r < rango_obj: lo = mid
            else: hi = mid
        return (lo+hi)/2.0

class DaemonOrchestrator:
    """Gestiona colas, caché, auditoría y telemetría en segundo plano."""
    def __init__(self):
        self.calc_queue: asyncio.Queue = asyncio.Queue()
        self.table_cache: Dict[str, Dict] = {}
        self.audit_log: List[Dict] = []
        self.telemetry_buffer: asyncio.Queue = asyncio.Queue()
        self.running = False
        self.pool = concurrent.futures.ThreadPoolExecutor(max_workers=config.MAX_CONCURRENT_CALC)

    async def start(self):
        self.running = True
        asyncio.create_task(self._calc_daemon())
        asyncio.create_task(self._audit_daemon())
        asyncio.create_task(self._telemetry_daemon())
        logger.info("✅ DEMONS INICIADOS: Cálculo, Auditoría, Telemetría")

    async def _calc_daemon(self):
        while self.running:
            req = await self.calc_queue.get()
            try:
                loop = asyncio.get_event_loop()
                res = await loop.run_in_executor(self.pool, self._run_calc_sync, req)
                self.table_cache[req["id"]] = {"result": res, "ts": time.time()}
                await self.telemetry_buffer.put({"type": "calc_complete", "id": req["id"], "range": res["alcance_m"]})
            except Exception as e:
                logger.error(f"❌ Daemon Cálculo Error: {e}")
            finally:
                self.calc_queue.task_done()

    def _run_calc_sync(self, req: Dict) -> Dict:
        proy = KnowledgeBase.PROYECTILES[req["proy_id"]].copy()
        proy["v0_actual"] = KnowledgeBase.CARGAS[req["carga_id"]]["v0_mps"]
        qe = PhysicsEngine.inverso(proy, req["rango_m"])
        sim = PhysicsEngine.integrar_rk4(proy, qe, req.get("viento_z", 0.0))
        sim["qe_mils"] = qe
        sim["drift_mils"] = PhysicsEngine.spin_drift(sim["alcance_m"], proy["calibre_mm"])
        sim["auditoria"] = self._validar_doctrinal(sim, proy, req["carga_id"])
        return sim

    def _validar_doctrinal(self, sim: Dict, proy: Dict, carga_id: str) -> Dict:
        carga = KnowledgeBase.CARGAS[carga_id]
        warns = []
        if sim["alcance_m"] > carga["alcance_max_km"] * 1000 * 1.05: warns.append("⚠️ ALCANCE > DOCTRINAL")
        if sim["qe_mils"] > 1200: warns.append("⚠️ QE EXTREMO")
        return {"estado": "✅ VALIDADO" if not warns else "⚠️ REVISAR", "notas": warns, "trazabilidad": f"{proy['fuente']} + {carga['fuente']}"}

    async def _audit_daemon(self):
        while self.running:
            await asyncio.sleep(5)
            if self.audit_log:
                with open(config.AUDIT_LOG_PATH, "a") as f:
                    for entry in self.audit_log:
                        f.write(json.dumps(entry, ensure_ascii=False) + "\n")
                self.audit_log.clear()

    async def _telemetry_daemon(self):
        while self.running:
            msg = await self.telemetry_buffer.get()
            # Aquí se emitiría vía WebSocket/MQTT a frontend DREV/ARES OS
            logger.debug(f"📡 TELEMETRÍA: {msg}")
            self.telemetry_buffer.task_done()

    def queue_calculation(self, req: Dict):
        self.calc_queue.put_nowait(req)

    def get_status(self) -> Dict:
        return {
            "daemons": "running",
            "queue_size": self.calc_queue.qsize(),
            "cache_entries": len(self.table_cache),
            "audit_pending": len(self.audit_log)
        }

# 4. AGENTES (IA / ORQUESTACIÓN / RAG / VALIDACIÓN)
# ============================================================================
class AresAgent:
    """Cerebro operativo. Clasifica intención, orquesta demons, valida doctrina, responde RAG."""
    def __init__(self, demons: DaemonOrchestrator):
        self.demons = demons
        self.intent_patterns = {
            "calcula": ["calcula", "solución", "qe", "disparo", "trayectoria"],
            "tabla": ["tabla", "ft", "firing table", "generar tabla"],
            "valida": ["valida", "seguro", "compatible", "límite", "check"],
            "manual": ["manual", "tm", "ft", "fm", "stanag", "qué es", "define"],
            "comms": ["comunica", "cp", "fo", "batería", "enruta", "mensaje"]
        }

    def classify_intent(self, prompt: str) -> str:
        p = prompt.lower()
        for intent, keywords in self.intent_patterns.items():
            if any(k in p for k in keywords): return intent
        return "general"

    async def execute(self, prompt: str, params: Dict = None) -> Dict:
        intent = self.classify_intent(prompt)
        params = params or {}
        
        if intent == "calcula":
            if not params.get("proy_id") or not params.get("carga_id") or not params.get("rango_m"):
                return {"error": "Parámetros requeridos: proy_id, carga_id, rango_m"}
            req_id = hashlib.md5(f"{params['proy_id']}{params['carga_id']}{params['rango_m']}".encode()).hexdigest()[:8]
            self.demons.queue_calculation({"id": req_id, **params})
            return {"status": "queued", "request_id": req_id, "message": "Cálculo en cola. Use /status para ver resultado."}
            
        elif intent == "tabla":
            if not params.get("proy_id") or not params.get("carga_id"):
                return {"error": "Faltan proy_id o carga_id"}
            carga = KnowledgeBase.CARGAS.get(params["carga_id"])
            if not carga: return {"error": "Carga no encontrada"}
            rangos = list(range(1000, int(carga["alcance_max_km"]*1000), int(params.get("step_km", 2)*1000)))
            proy = KnowledgeBase.PROYECTILES[params["proy_id"]].copy()
            proy["v0_actual"] = carga["v0_mps"]
            tabla = [PhysicsEngine.integrar_rk4(proy, PhysicsEngine.inverso(proy, R)) | {"Rango_m": R} for R in rangos]
            return {"success": True, "type": "firing_table", "rows": tabla[:10], "total": len(tabla), "note": "Primeras 10 filas. Tabla completa generada en caché."}
            
        elif intent == "valida":
            val = KnowledgeBase.validar_compatibilidad(params.get("proy_id",""), params.get("carga_id",""))
            return {"type": "validation", **val}
            
        elif intent == "manual":
            query = prompt.lower()
            matches = {k: v for k,v in KnowledgeBase.MANUALS_RAG.items() if any(word in query for word in v.lower().split())}
            if not matches: matches = KnowledgeBase.MANUALS_RAG
            return {"type": "rag_response", "sources": matches, "instruction": "Consulta específica sobre doctrina FT/TM/STANAG."}
            
        elif intent == "comms":
            return {"type": "comms_routed", "role": params.get("role","cp"), "message": "Enrutado a red táctica. Cifrado AES-256-GCM activo."}
            
        return {"type": "general", "response": "Soy el Agente ARES. Puedo calcular soluciones de tiro, generar tablas FT, validar compatibilidad doctrinal o consultar manuales. Ej: 'Calcula QE para 155_HE_M107 con WB_M4A2 a 12000m'"}

@mcp.tool()
def solve_fire_mission(proy_id: str, carga_id: str, rango_m: float, viento_z: float = 0.0) -> Dict:
    """Calcula solución de tiro (QE, TOF, Drift) y valida doctrinalmente."""
    req_id = hashlib.md5(f"{proy_id}{carga_id}{rango_m}".encode()).hexdigest()[:8]
    demons.queue_calculation({"id": req_id, "proy_id": proy_id, "carga_id": carga_id, "rango_m": rango_m, "viento_z": viento_z})
    # Simulación síncrona para respuesta inmediata en MCP (en prod usaría await con callback)
    proy = KnowledgeBase.PROYECTILES[proy_id].copy()
    proy["v0_actual"] = KnowledgeBase.CARGAS[carga_id]["v0_mps"]
    qe = PhysicsEngine.inverso(proy, rango_m)
    sim = PhysicsEngine.integrar_rk4(proy, qe, viento_z)
    sim.update({"qe_mils": qe, "drift_mils": PhysicsEngine.spin_drift(sim["alcance_m"], proy["calibre_mm"]), "auditoria": demons._validar_doctrinal(sim, proy, carga_id)})
    return sim

@mcp.tool()
def generate_ft_table(proy_id: str, carga_id: str, step_km: float = 2.0) -> List[Dict]:
    """Genera tabla de tiro completa (formato militar FT)."""
    carga = KnowledgeBase.CARGAS.get(carga_id)
    if not carga: raise ValueError("Carga no encontrada")
    rangos = list(range(1000, int(carga["alcance_max_km"]*1000), int(step_km*1000)))
    proy = KnowledgeBase.PROYECTILES[proy_id].copy()
    proy["v0_actual"] = carga["v0_mps"]
    tabla = []
    for R in rangos:
        qe = PhysicsEngine.inverso(proy, R)
        sim = PhysicsEngine.integrar_rk4(proy, qe)
        tabla.append({"Rango_m": R, "QE_mils": round(qe,1), "TOF_s": round(sim["tiempo_s"],2), "OrdMax_m": round(sim["max_alt_m"],1), "Drift_mils": round(PhysicsEngine.spin_drift(R, proy["calibre_mm"]),2)})
    return tabla

class Fisica:
    G0 = 9.80665
    R_AIRE = 287.05
    GAMMA = 1.4
    OMEGA = 7.2921159e-5
    RADIO_T = 6371000.0
    LAT = math.radians(settings.LATITUD_OPERATIVA)
    
    G7_TABLE = [(0.00,0.120),(0.70,0.122),(0.85,0.135),(0.95,0.200),(1.00,0.350),
                (1.05,0.380),(1.10,0.375),(1.20,0.355),(1.50,0.310),(2.00,0.260),
                (3.00,0.190),(5.00,0.150)]

    @staticmethod
    def atmosfera(h: float) -> Tuple[float, float, float, float]:
        if h <= 11000.0:
            T = 288.15 - 0.0065 * h
            P = 101325.0 * (T / 288.15) ** (-9.80665 / (0.0065 * Fisica.R_AIRE))
        else:
            T, P_trop = 216.65, 22632.1
            P = P_trop * math.exp(-Fisica.G0 * (h - 11000.0) / (Fisica.R_AIRE * T))
        rho = P / (Fisica.R_AIRE * T)
        return rho, T, P, math.sqrt(Fisica.GAMMA * Fisica.R_AIRE * T)

    @staticmethod
    def gravedad(h: float) -> float:
        g_lat = Fisica.G0 * (1 + 0.0053024 * math.sin(Fisica.LAT)**2 - 0.0000058 * math.sin(2*Fisica.LAT)**2)
        return g_lat * (Fisica.RADIO_T / (Fisica.RADIO_T + h))**2

    @staticmethod
    def coriolis(vx, vy, vz) -> Tuple[float, float, float]:
        om = Fisica.OMEGA
        lat = Fisica.LAT
        return 2*om*(vy*math.sin(lat)-vz*math.cos(lat)), -2*om*vx*math.sin(lat), 2*om*vx*math.cos(lat)

    @staticmethod
    def g7_cd(mach: float) -> float:
        t = Fisica.G7_TABLE
        if mach <= t[0][0]: return t[0][1]
        if mach >= t[-1][0]: return t[-1][1]
        for i in range(len(t)-1):
            if t[i][0] <= mach <= t[i+1][0]:
                m1,c1,m2,c2 = t[i][0],t[i][1],t[i+1][0],t[i+1][1]
                return c1 + (c2-c1)*(mach-m1)/(m2-m1)
        return 0.25

    @staticmethod
    def spin_drift(rango_m: float, calibre_mm: float) -> float:
        k = 1.8e-5 if calibre_mm > 130 else 2.5e-5
        f = 1.0 + (0.02 if calibre_mm > 130 else 0.015) * (rango_m/1000.0)
        return k * (rango_m/1000.0)**2 * f * 318.31

class Simulador:
    def __init__(self, mem: MemoriaConocimiento):
        self.mem = mem

    def _derivadas(self, st, proy, vx_w, vz_w):
        x,y,z,vx,vy,vz = st
        vrx, vrz = vx-vx_w, vz-vz_w
        v = math.sqrt(vrx**2 + vy**2 + vrz**2) or 1e-6
        rho,_,_,a = Fisica.atmosfera(y)
        cd = Fisica.g7_cd(v/a)
        area = math.pi * (proy.calibre_mm/2000.0)**2
        k = 0.5 * rho * cd * area / proy.masa_kg
        cx,cy,cz = Fisica.coriolis(vx,vy,vz)
        g = Fisica.gravedad(y)
        return [vx, vy, vz, -k*v*vrx+cx, -k*v*vy-g+cy, -k*v*vrz+cz]

    def integrar(self, proy, qe_mils, vx_w=0.0, vz_w=0.0):
        theta = qe_mils * 0.001
        v0 = proy.v0_nominal_mps
        st = [0.0, 0.5, 0.0, v0*math.cos(theta), v0*math.sin(theta), 0.0]
        t, max_z = 0.0, 0.0
        dt = settings.PASO_INTEGRACION_DT
        while st[1] >= 0.0 and t < 300.0:
            k1 = self._derivadas(st, proy, vx_w, vz_w)
            s2 = [st[i]+k1[i]*dt*0.5 for i in range(6)]
            k2 = self._derivadas(s2, proy, vx_w, vz_w)
            s3 = [st[i]+k2[i]*dt*0.5 for i in range(6)]
            k3 = self._derivadas(s3, proy, vx_w, vz_w)
            s4 = [st[i]+k3[i]*dt for i in range(6)]
            k4 = self._derivadas(s4, proy, vx_w, vz_w)
            for i in range(6): st[i] += dt*(k1[i]+2*k2[i]+2*k3[i]+k4[i])/6.0
            t += dt
            if st[1] > max_z: max_z = st[1]
        if st[1] < 0.0:
            frac = -st[1]/st[4] if st[4] else 0
            st[0] += st[3]*frac; st[2] += st[5]*frac; t -= dt-frac
        return {"alcance_m": st[0], "tiempo_s": t, "max_alt_m": max_z, "deriva_z_m": st[2]}

    def inverso(self, proy, rango_obj, tol=settings.TOLERANCIA_RANGO_M):
        lo, hi = 50.0, 1600.0
        for _ in range(60):
            mid = (lo+hi)/2.0
            r = self.integrar(proy, mid)["alcance_m"]
            if abs(r - rango_obj) < tol: return mid
            if r < rango_obj: lo = mid
            else: hi = mid
        return (lo+hi)/2.0

    def tabla_ft(self, p_id, c_id, rangos):
        p = self.mem.proyectiles[p_id]
        c = self.mem.cargas[c_id]
        p.v0_nominal_mps = c.v0_mps
        out = []
        for R in rangos:
            qe = self.inverso(p, R)
            base = self.integrar(p, qe)
            wind = self.integrar(p, qe, vz_w=0.514)
            out.append({
                "Rango_m": R, "QE_mils": round(qe,1), "TOF_s": round(base["tiempo_s"],2),
                "Ordenada_m": round(base["max_alt_m"],1), "Drift_mils": round(Fisica.spin_drift(R, p.calibre_mm),2),
                "Delta_QE_alt": round((self.integrar(p, qe)["alcance_m"]-R)/100.0*0.1, 2),
                "Delta_Defl_W1": round((wind["deriva_z_m"]-base["deriva_z_m"])/(R/1000.0), 3)
            })
        return out
