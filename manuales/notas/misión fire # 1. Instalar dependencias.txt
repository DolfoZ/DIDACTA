class Fisica:
    G0 = 9.80665
    R_AIRE = 287.05
    GAMMA = 1.4
    OMEGA = 7.2921159e-5
    RADIO_T = 6371000.0
    LAT = math.radians(settings.LATITUD_OPERATIVA)
    
    G7_TABLE = [(0.00,0.120),(0.70,0.122),(0.85,0.135),(0.95,0.200),(1.00,0.350),
                (1.05,0.380),(1.10,0.375),(1.20,0.355),(1.50,0.310),(2.00,0.260),
                (3.00,0.190),(5.00,0.150)]

    @staticmethod
    def atmosfera(h: float) -> Tuple[float, float, float, float]:
        if h <= 11000.0:
            T = 288.15 - 0.0065 * h
            P = 101325.0 * (T / 288.15) ** (-9.80665 / (0.0065 * Fisica.R_AIRE))
        else:
            T, P_trop = 216.65, 22632.1
            P = P_trop * math.exp(-Fisica.G0 * (h - 11000.0) / (Fisica.R_AIRE * T))
        rho = P / (Fisica.R_AIRE * T)
        return rho, T, P, math.sqrt(Fisica.GAMMA * Fisica.R_AIRE * T)

    @staticmethod
    def gravedad(h: float) -> float:
        g_lat = Fisica.G0 * (1 + 0.0053024 * math.sin(Fisica.LAT)**2 - 0.0000058 * math.sin(2*Fisica.LAT)**2)
        return g_lat * (Fisica.RADIO_T / (Fisica.RADIO_T + h))**2

    @staticmethod
    def coriolis(vx, vy, vz) -> Tuple[float, float, float]:
        om = Fisica.OMEGA
        lat = Fisica.LAT
        return 2*om*(vy*math.sin(lat)-vz*math.cos(lat)), -2*om*vx*math.sin(lat), 2*om*vx*math.cos(lat)

    @staticmethod
    def g7_cd(mach: float) -> float:
        t = Fisica.G7_TABLE
        if mach <= t[0][0]: return t[0][1]
        if mach >= t[-1][0]: return t[-1][1]
        for i in range(len(t)-1):
            if t[i][0] <= mach <= t[i+1][0]:
                m1,c1,m2,c2 = t[i][0],t[i][1],t[i+1][0],t[i+1][1]
                return c1 + (c2-c1)*(mach-m1)/(m2-m1)
        return 0.25

    @staticmethod
    def spin_drift(rango_m: float, calibre_mm: float) -> float:
        k = 1.8e-5 if calibre_mm > 130 else 2.5e-5
        f = 1.0 + (0.02 if calibre_mm > 130 else 0.015) * (rango_m/1000.0)
        return k * (rango_m/1000.0)**2 * f * 318.31

class Simulador:
    def __init__(self, mem: MemoriaConocimiento):
        self.mem = mem

    def _derivadas(self, st, proy, vx_w, vz_w):
        x,y,z,vx,vy,vz = st
        vrx, vrz = vx-vx_w, vz-vz_w
        v = math.sqrt(vrx**2 + vy**2 + vrz**2) or 1e-6
        rho,_,_,a = Fisica.atmosfera(y)
        cd = Fisica.g7_cd(v/a)
        area = math.pi * (proy.calibre_mm/2000.0)**2
        k = 0.5 * rho * cd * area / proy.masa_kg
        cx,cy,cz = Fisica.coriolis(vx,vy,vz)
        g = Fisica.gravedad(y)
        return [vx, vy, vz, -k*v*vrx+cx, -k*v*vy-g+cy, -k*v*vrz+cz]

    def integrar(self, proy, qe_mils, vx_w=0.0, vz_w=0.0):
        theta = qe_mils * 0.001
        v0 = proy.v0_nominal_mps
        st = [0.0, 0.5, 0.0, v0*math.cos(theta), v0*math.sin(theta), 0.0]
        t, max_z = 0.0, 0.0
        dt = settings.PASO_INTEGRACION_DT
        while st[1] >= 0.0 and t < 300.0:
            k1 = self._derivadas(st, proy, vx_w, vz_w)
            s2 = [st[i]+k1[i]*dt*0.5 for i in range(6)]
            k2 = self._derivadas(s2, proy, vx_w, vz_w)
            s3 = [st[i]+k2[i]*dt*0.5 for i in range(6)]
            k3 = self._derivadas(s3, proy, vx_w, vz_w)
            s4 = [st[i]+k3[i]*dt for i in range(6)]
            k4 = self._derivadas(s4, proy, vx_w, vz_w)
            for i in range(6): st[i] += dt*(k1[i]+2*k2[i]+2*k3[i]+k4[i])/6.0
            t += dt
            if st[1] > max_z: max_z = st[1]
        if st[1] < 0.0:
            frac = -st[1]/st[4] if st[4] else 0
            st[0] += st[3]*frac; st[2] += st[5]*frac; t -= dt-frac
        return {"alcance_m": st[0], "tiempo_s": t, "max_alt_m": max_z, "deriva_z_m": st[2]}

    def inverso(self, proy, rango_obj, tol=settings.TOLERANCIA_RANGO_M):
        lo, hi = 50.0, 1600.0
        for _ in range(60):
            mid = (lo+hi)/2.0
            r = self.integrar(proy, mid)["alcance_m"]
            if abs(r - rango_obj) < tol: return mid
            if r < rango_obj: lo = mid
            else: hi = mid
        return (lo+hi)/2.0

    def tabla_ft(self, p_id, c_id, rangos):
        p = self.mem.proyectiles[p_id]
        c = self.mem.cargas[c_id]
        p.v0_nominal_mps = c.v0_mps
        out = []
        for R in rangos:
            qe = self.inverso(p, R)
            base = self.integrar(p, qe)
            wind = self.integrar(p, qe, vz_w=0.514)
            out.append({
                "Rango_m": R, "QE_mils": round(qe,1), "TOF_s": round(base["tiempo_s"],2),
                "Ordenada_m": round(base["max_alt_m"],1), "Drift_mils": round(Fisica.spin_drift(R, p.calibre_mm),2),
                "Delta_QE_alt": round((self.integrar(p, qe)["alcance_m"]-R)/100.0*0.1, 2),
                "Delta_Defl_W1": round((wind["deriva_z_m"]-base["deriva_z_m"])/(R/1000.0), 3)
            })
        return out

# ==================== ALCANCE 3: AUDITORÍA & PERSISTENCIA ====================
class Auditoria:
    def __init__(self, mem: MemoriaConocimiento, sim: Simulador):
        self.mem = mem
        self.sim = sim
        Path(settings.LOG_AUDIT_PATH).parent.mkdir(parents=True, exist_ok=True)

    def ejecutar_tiro(self, p_id: str, c_id: str, rango: float, viento_z: float = 0.0) -> dict:
        if p_id not in self.mem.proyectiles or c_id not in self.mem.cargas:
            raise ValueError("Munición o carga no registrada en memoria doctrinal.")
        if c_id not in self.mem.proyectiles[p_id].compatible_cargas:
            raise ValueError(f"Combinación {p_id} + {c_id} NO COMPATIBLE según FT oficiales.")
        
        proy = self.mem.proyectiles[p_id]
        carga = self.mem.cargas[c_id]
        proy.v0_nominal_mps = carga.v0_mps
        
        qe = self.sim.inverso(proy, rango)
        datos = self.sim.integrar(proy, qe, vz_w=viento_z)
        drift = Fisica.spin_drift(datos["alcance_m"], proy.calibre_mm)
        
        # Auditoría doctrinal
        warns = []
        if datos["alcance_m"] > carga.alcance_max_km * 1000 * 1.05:
            warns.append("⚠️ ALCANCE SUPERIOR AL DOCTRINAL")
        if qe > 1200: warns.append("⚠️ QE > 1200 mils (Verificar máscara)")
        
        resultado = {
            "timestamp": datetime.utcnow().isoformat(),
            "proyectil": proy.nombre, "carga": carga.designacion,
            "rango_objetivo": rango, "alcance_real": round(datos["alcance_m"],1),
            "qe_mils": round(qe,1), "tof_s": round(datos["tiempo_s"],2),
            "drift_mils": round(drift,2), "v0_aplicada": carga.v0_mps,
            "auditoria": {"estado": "✅ VALIDADO" if not warns else "⚠️ CON ADVERTENCIAS", "notas": warns}
        }
        
        # Persistencia segura
        with open(settings.LOG_AUDIT_PATH, "a", encoding="utf-8") as f:
            f.write(json.dumps(resultado, ensure_ascii=False) + "\n")
            
        return resultado

class FireRequest(BaseModel):

import asyncio

app = FastAPI(title="ARES Ballistic Core API", version="2.1.0")
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

# Dependency Injection Singleton
def get_auditor():
    mem = MemoriaConocimiento()
    return Auditoria(mem, Simulador(mem))

@app.get("/health")
async def health():
    return {"status": "operational", "engine": "loaded", "docs": "/docs"}

@app.post("/v1/fire-solution", response_model=FireResponse)
async def fire_solution(req: FireRequest, auditor: Auditoria = Depends(get_auditor)):
    try:
        # Run sync math in thread pool to avoid blocking event loop
        loop = asyncio.get_event_loop()
        res = await loop.run_in_executor(None, auditor.ejecutar_tiro, 
                                         req.projectile_id, req.charge_id, req.range_m, req.wind_z_mps)
        return FireResponse(success=True, data=res)
    except ValueError as e:
        return FireResponse(success=False, error=str(e))
    except Exception as e:
        raise HTTPException(500, detail=f"Simulation error: {str(e)}")

@app.post("/v1/generate-table")
async def generate_table(req: TableRequest, auditor: Auditoria = Depends(get_auditor)):
    try:
        carga = auditor.mem.cargas[req.charge_id]
        rangos = list(range(1000, int(carga.alcance_max_km*1000), int(req.step_km*1000)))
        loop = asyncio.get_event_loop()
        table = await loop.run_in_executor(None, auditor.sim.tabla_ft, req.projectile_id, req.charge_id, rangos)
        return {"success": True, "rows": table}
    except Exception as e:
        raise HTTPException(400, detail=str(e))
from mcp.server.fastmcp import FastMCP
from core.engine import MemoriaConocimiento, Simulador, Auditoria

mcp = FastMCP("ares-ballistic-brain")

# Shared engine instance
mem = MemoriaConocimiento()

@mcp.tool()
def calculate_fire_solution(projectile_id: str, charge_id: str, range_m: float, wind_z: float = 0.0) -> dict:
    """Calcula QE, TOF, Drift y auditoría doctrinal para un disparo específico."""
    try:
        return audit.ejecutar_tiro(projectile_id, charge_id, range_m, wind_z)
    except ValueError as e:
        return {"error": str(e)}

@mcp.tool()
def generate_firing_table(projectile_id: str, charge_id: str, step_km: float = 2.0) -> list:
    """Genera tabla de tiro completa (7 columnas) según formato FT militar."""
    try:
        carga = audit.mem.cargas[charge_id]
        rangos = list(range(1000, int(carga.alcance_max_km*1000), int(step_km*1000)))
        return sim.tabla_ft(projectile_id, charge_id, rangos)
    except Exception as e:
        return [{"error": str(e)}]

@mcp.tool()
def check_compatibility(projectile_id: str, charge_id: str) -> dict:
    """Valida compatibilidad doctrinal entre proyectil y carga propulsora."""
    p = audit.mem.proyectiles.get(projectile_id)
    c = audit.mem.cargas.get(charge_id)
    if not p or not c:
        return {"compatible": False, "reason": "ID no encontrado en memoria"}
    return {"compatible": charge_id in p.compatible_cargas, "sources": "FT 155-AM-2 / TM 43-0001-28"}

@mcp.resource("ares://memory/projectiles")
def get_projectiles() -> str:
    """Lista completa de proyectiles indexados y validados."""
    return str({k: v.__dict__ for k,v in audit.mem.proyectiles.items()})

@mcp.resource("ares://physics/constants")
def get_constants() -> str:
    """Constantes físicas y tabla G7 utilizadas por el motor."""
    return f"G0={Fisica.G0}, OMEGA={Fisica.OMEGA}, G7_TABLE={Fisica.G7_TABLE}"
