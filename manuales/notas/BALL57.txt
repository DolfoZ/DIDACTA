    def _get_recommended_drag(self, proj_name):
        """Devuelve el modelo G recomendado según el nombre del proyectil."""
        name_upper = proj_name.upper()
        # Proyectiles de bajo arrastre / alcance extendido → G7
        if any(x in name_upper for x in ['HERA', 'RAP', 'M549', 'M864', 'RAAM', 'DPICM', 'M795']):
            return 'G7'
        # Proyectiles estándar de base plana → G1
        elif any(x in name_upper for x in ['M107', 'M106', 'HE M1', 'ILLUM', 'SMOKE', 'HC', 'M1 ']):
            return 'G1'
        else:
            return 'G1'  # por defecto conservador

    def analyze_formula_target(self):
        """Analiza qué variables modificar para alcanzar los objetivos"""
        if self.trajectory_data is None:
            messagebox.showwarning("Sin datos", "Ejecuta una simulación primero")
            return
        
        data = self.trajectory_data
        y = data["y"]
        t = data["t"]
        current_range = np.sqrt(y[0, -1]**2 + y[1, -1]**2)
        current_time = t[-1]
        current_defl = y[1, -1]
        
        v0 = self.current_projectile['muzzle_velocity']
        elev = self.elevation.get()
        lat = self.latitude.get()
        
        analysis_text = ""
        self.suggested_values = {}
        
        # Analizar alcance
        if self.target_range.get():
            target = float(self.target_range.get())
            
            a_actual = self.formula_coeffs['range']['a']
            b_actual = self.formula_coeffs['range']['b']
            denominador = (v0 ** a_actual) * (np.sin(2 * np.radians(elev)) ** b_actual)
            k1_needed = target / denominador
            
            analysis_text += f"ALCANCE OBJETIVO: {target:.0f} m (actual: {current_range:.0f} m)\n"
            analysis_text += "-" * 40 + "\n"
            
            self.suggested_values['range_k'] = k1_needed
            analysis_text += f"• k₁ (Factor de escala): {k1_needed:.8f}\n"
            analysis_text += f"• a (Exponente velocidad): {a_actual:.3f} (mantener)\n"
            analysis_text += f"• b (Exponente angular): {b_actual:.3f} (mantener)\n\n"
            
            k1_temp = k1_needed
            a_temp = a_actual
            b_temp = b_actual
        else:
            k1_temp = self.formula_coeffs['range']['k']
            a_temp = self.formula_coeffs['range']['a']
            b_temp = self.formula_coeffs['range']['b']
        
        # Analizar tiempo
        if self.target_time.get():
            target = float(self.target_time.get())
            
            c_actual = self.formula_coeffs['time']['c']
            d_actual = self.formula_coeffs['time']['d']
            denominador = (v0 ** c_actual) * (np.sin(np.radians(elev)) ** d_actual)
            k2_needed = target / denominador
            
            analysis_text += f"TIEMPO OBJETIVO: {target:.2f} s (actual: {current_time:.2f} s)\n"
            analysis_text += "-" * 40 + "\n"
            
            self.suggested_values['time_k'] = k2_needed
            analysis_text += f"• k₂ (Factor escala temporal): {k2_needed:.6f}\n"
            analysis_text += f"• c (Exponente velocidad): {c_actual:.3f} (mantener)\n"
            analysis_text += f"• d (Exponente angular): {d_actual:.3f} (mantener)\n\n"
            
            k2_temp = k2_needed
            c_temp = c_actual
            d_temp = d_actual
        else:
            k2_temp = self.formula_coeffs['time']['k']
            c_temp = self.formula_coeffs['time']['c']
            d_temp = self.formula_coeffs['time']['d']
        
        # Analizar deflexión
        if self.target_deflection.get():
            target = float(self.target_deflection.get())
            
            e_actual = self.formula_coeffs['deflection']['e']
            omega = 7.2921159e-5
            v_avg = current_range / current_time
            e_needed = (2 * omega * np.sin(np.radians(lat)) * v_avg * (current_time ** 2)) / target
            
            analysis_text += f"DEFLEXIÓN OBJETIVO: {target:.2f} m (actual: {current_defl:.2f} m)\n"
            analysis_text += "-" * 40 + "\n"
            
            self.suggested_values['defl_e'] = e_needed
            analysis_text += f"• e (Amortiguamiento Coriolis): {e_needed:.3f}\n"
            analysis_text += f"  Nota: ↑e = ↓deflexión, ↓e = ↑deflexión\n\n"
            
            e_temp = e_needed
        else:
            e_temp = self.formula_coeffs['deflection']['e']
        
        if not analysis_text:
            analysis_text = "Ingrese un valor objetivo en los campos superiores.\n"
        
        analysis_text += "\n" + "=" * 45 + "\n"
        analysis_text += "Presione 'APLICAR VALORES SUGERIDOS' para usarlos.\n"
        
        self.temp_k1 = k1_temp
        self.temp_a = a_temp
        self.temp_b = b_temp
        self.temp_k2 = k2_temp
        self.temp_c = c_temp
        self.temp_d = d_temp
        self.temp_e = e_temp
        
        self.analysis_result.config(state="normal")
        self.analysis_result.delete(1.0, tk.END)
        self.analysis_result.insert(1.0, analysis_text)
        self.analysis_result.config(state="disabled")


    def apply_suggested_values(self):
        """Aplica los valores sugeridos a los campos de entrada"""
        if not self.suggested_values:
            messagebox.showwarning("Sin sugerencias", "Primero ejecute 'CALCULAR CÓMO LLEGAR'")
            return
        
        for key, value in self.suggested_values.items():
            if key == 'range_k':
                self.coeff_range_k.delete(0, tk.END)
                self.coeff_range_k.insert(0, f"{value:.8f}")
            elif key == 'range_a':
                self.coeff_range_a.delete(0, tk.END)
                self.coeff_range_a.insert(0, f"{value:.3f}")
            elif key == 'range_b':
                self.coeff_range_b.delete(0, tk.END)
                self.coeff_range_b.insert(0, f"{value:.3f}")
            elif key == 'time_k':
                self.coeff_time_k.delete(0, tk.END)
                self.coeff_time_k.insert(0, f"{value:.6f}")
            elif key == 'time_c':
                self.coeff_time_c.delete(0, tk.END)
                self.coeff_time_c.insert(0, f"{value:.3f}")
            elif key == 'time_d':
                self.coeff_time_d.delete(0, tk.END)
                self.coeff_time_d.insert(0, f"{value:.3f}")
            elif key == 'defl_e':
                self.coeff_defl_e.delete(0, tk.END)
                self.coeff_defl_e.insert(0, f"{value:.3f}")
        
        self.update_formula_preview()
        messagebox.showinfo("Éxito", "Valores sugeridos aplicados a los campos.\nPresione 'APLICAR FÓRMULAS' para ver los resultados.")
    

    def update_formula_preview(self):
        """Actualiza las previsualizaciones de fórmulas"""
        if not self.trajectory_data:
            return
        
        try:
            v0 = self.current_projectile['muzzle_velocity']
            elev = self.elevation.get()
            lat = self.latitude.get()
            data = self.trajectory_data
            y = data["y"]
            t = data["t"]
            
            current_range = np.sqrt(y[0, -1]**2 + y[1, -1]**2)
            current_time = t[-1]
            current_defl = y[1, -1]
            current_alt = np.max(-y[2, :])
            
            k1 = float(self.coeff_range_k.get() or self.formula_coeffs['range']['k'])
            a = float(self.coeff_range_a.get() or self.formula_coeffs['range']['a'])
            b = float(self.coeff_range_b.get() or self.formula_coeffs['range']['b'])
            
            k2 = float(self.coeff_time_k.get() or self.formula_coeffs['time']['k'])
            c = float(self.coeff_time_c.get() or self.formula_coeffs['time']['c'])
            d = float(self.coeff_time_d.get() or self.formula_coeffs['time']['d'])
            
            e = float(self.coeff_defl_e.get() or self.formula_coeffs['deflection']['e'])
            
            k3 = float(self.coeff_alt_k.get() or self.formula_coeffs['altitude']['k'])
            f = float(self.coeff_alt_f.get() or self.formula_coeffs['altitude']['f'])
            g = float(self.coeff_alt_g.get() or self.formula_coeffs['altitude']['g'])
            
            range_new = k1 * (v0 ** a) * (np.sin(2 * np.radians(elev)) ** b)
            time_new = k2 * (v0 ** c) * (np.sin(np.radians(elev)) ** d)
            omega = 7.2921159e-5
            v_avg = range_new / time_new if time_new > 0 else 0
            defl_new = 2 * omega * np.sin(np.radians(lat)) * v_avg * (time_new ** 2) / e
            alt_new = k3 * (v0 ** f) * (np.sin(np.radians(elev)) ** (2 * g))
            
            self.range_k_effect.config(text=f"(cambio: {range_new - current_range:+.1f} m)")
            self.range_a_effect.config(text=f"(sensibilidad: ~{(range_new/current_range - 1)*100:+.1f}%)")
            self.range_b_effect.config(text=f"(eficiencia angular)")
            self.range_preview.config(text=f"→ Alcance estimado: {range_new:.1f} m")
            
            self.time_k_effect.config(text=f"(cambio: {time_new - current_time:+.2f} s)")
            self.time_c_effect.config(text=f"(sensibilidad: ~{(time_new/current_time - 1)*100:+.1f}%)")
            self.time_d_effect.config(text=f"(eficiencia angular)")
            self.time_preview.config(text=f"→ Tiempo estimado: {time_new:.2f} s")
            
            self.defl_e_effect.config(text=f"(cambio: {defl_new - current_defl:+.2f} m)")
            self.defl_preview.config(text=f"→ Deflexión estimada: {defl_new:.2f} m")
            
            self.alt_k_effect.config(text=f"(cambio: {alt_new - current_alt:+.1f} m)")
            self.alt_f_effect.config(text=f"(sensibilidad: ~{(alt_new/current_alt - 1)*100:+.1f}%)")
            self.alt_g_effect.config(text=f"(eficiencia angular)")
            self.alt_preview.config(text=f"→ Altitud estimada: {alt_new:.1f} m")
            
        except Exception:
            pass

    def simulate_initial(self):

            final = y[:, -1]
            max_alt = np.max(-y[2, :])
            range_val = np.sqrt(final[0]**2 + final[1]**2)

                'impact_vel': impact_vel,
                'mach': mach_val,
                'final_alt': final_alt
            }
            self.update_results_panel()  # ← NUEVO MÉTODO
            self.update_formula_preview()
    


    def simulate(self):
        """Prepara y ejecuta la simulación"""
        try:
            if not self.current_projectile:
                messagebox.showerror("Error", "No hay proyectil cargado")
                return
            
            # Aplicar offset de temperatura
            self.atmosphere.set_temperature_offset(self.temp_offset.get())
            self.atmosphere.set_humidity(self.humidity.get())
            self.atmosphere.set_co2(self.co2_ppm.get())
            
            self._run_simulation(
                self.current_projectile["caliber"],
                self.current_projectile["caliber_unit"],
                self.current_projectile["drag_function"],
                self.current_projectile["ballistic_coefficient"],
                self.current_projectile["mass"],
                self.current_projectile["muzzle_velocity"],
                self.elevation.get(),
                self.azimuth.get()
            )          
        
        except Exception as e:
            messagebox.showerror("Error en simulación", str(e))
    

    def _run_simulation(self, caliber, caliber_unit, drag_func, bc, mass, v0, elev, az):
        """Ejecuta la simulación numérica con todos los modelos físicos"""
        from src.geophysics.coriolis import Coriolis
        from src.atmosphere.wind import WindModel
        from src.aerodynamics.magnus import MagnusEffect
        
        geom = ProjectileGeometry(caliber, caliber_unit)
        selected_drag = self.drag_function.get()  # Usar el valor del Combobox
        drag = DragCoefficient(selected_drag, bc)
        coriolis = Coriolis(self.latitude.get())
        wind = WindModel(self.wind_speed.get(), self.wind_direction.get())
        
        magnus = None
        if self.use_magnus.get():
            spin_rate = self.spin_rate_rpm.get()
            # Protección contra RPM extremadamente bajas que causan inestabilidad
            if spin_rate < 1000:
                spin_rate = 15000  # Valor razonable para artillería
                self.spin_rate_rpm.set(spin_rate)  # Actualizar la GUI
            
            magnus = MagnusEffect(
                spin_rate_rpm=spin_rate,
                caliber_m=caliber / 1000.0,
                rifling_twist_cal=self.rifling_twist.get()
            )
            if spin_rate == 0:
                magnus.calculate_spin_rate(v0)
        
        forces = Forces(drag, geom, mass, coriolis_model=coriolis, 
                       latitude=self.latitude.get(), wind_model=wind,
                       magnus_model=magnus)
       
        def derivatives(t, y):
            vel = y[3:6]
            f = forces.calculate(t, y)
            acc = f / mass
            return np.concatenate([vel, acc])
        
        is_base_ejection = False  # ← INICIALIZACIÓN
        # Definir evento de detonación según tipo de espoleta
        if self.fuze_type.get() == "PDA":
            # Espoleta PDA: detonación en impacto con el suelo (z = 0)
            def fuze_event(t, y):
                return -y[2]
            fuze_event.terminal = True
            fuze_event.direction = -1
            self.tia_expulsion_point = None  # No hay punto de expulsión
            
        else:  # TIA
            t_prog = self.fuze_time.get()
            if t_prog <= 0:
                messagebox.showerror("Error", "El ajuste de tiempo TIA debe ser mayor a 0 segundos.")
                self.trajectory_data = None
                return
            
            # Determinar si es proyectil de carga base (ILLUM, SMOKE, RAAM)
            proj_name = self.current_projectile.get('name', '').upper()
            is_base_ejection = any(x in proj_name for x in ['ILLUM', 'SMOKE', 'RAAM', 'M314', 'M485', 'HC'])
            
            if is_base_ejection:
                # Proyectil de carga base: TIA marca expulsión, pero el cuerpo continúa
                def tia_expulsion_event(t, y):
                    return t - t_prog
                tia_expulsion_event.terminal = False  # NO detiene la simulación
                tia_expulsion_event.direction = 0
                
                # Guardar punto de expulsión para marcarlo en gráficos
                self.tia_expulsion_point = None
                
                # El evento terminal sigue siendo el impacto
                def fuze_event(t, y):
                    return -y[2]
                fuze_event.terminal = True
                fuze_event.direction = -1
                
                # Usar ambos eventos
                events = [tia_expulsion_event, fuze_event]
            else:
                # Proyectil HE: TIA causa detonación en aire (terminal)
                def fuze_event(t, y):
                    return t - t_prog
                fuze_event.terminal = True
                fuze_event.direction = 0
                events = fuze_event
                self.tia_expulsion_point = None
        
        transform = RotationTransform(az, elev)
        v0_vec = transform.muzzle_to_ned(v0)
        y0 = np.array([0.0, 0.0, 0.0, v0_vec[0], v0_vec[1], v0_vec[2]])
        
        try:
            if is_base_ejection and self.fuze_type.get() == "TIA":
                result = solve_ivp(
                    derivatives, (0, 300), y0,
                    method="DOP853", events=events,
                    rtol=1e-10, atol=1e-12, max_step=0.5
                )
                # Encontrar punto de expulsión
                if len(result.t_events) > 0 and len(result.t_events[0]) > 0:
                    t_exp = result.t_events[0][0]
                    # Interpolar posición en t_exp
                    from scipy.interpolate import interp1d
                    x_interp = interp1d(result.t, result.y[0])(t_exp)
                    y_interp = interp1d(result.t, result.y[1])(t_exp)
                    z_interp = interp1d(result.t, result.y[2])(t_exp)
                    self.tia_expulsion_point = (x_interp, y_interp, z_interp)
            else:
                result = solve_ivp(
                    derivatives, (0, 300), y0,
                    method="DOP853", events=fuze_event,
                    rtol=1e-10, atol=1e-12, max_step=0.5
                )
        except Exception as e:
            messagebox.showerror("Error en la integración", 
                                 f"La simulación numérica falló:\n{str(e)}\n\n"
                                 "Posibles causas:\n"
                                 "- Magnus con RPM muy bajo/altas\n"
                                 "- Viento extremo\n"
                                 "- Combinación de parámetros inestable")
            self.trajectory_data = None
            return
        
        if not result.success:
            messagebox.showwarning("Simulación incompleta", 
                                   f"La integración no convergió correctamente.\n{result.message}")
        
        self.trajectory_data = {"t": result.t, "y": result.y}
        self.update_plots()
        self.update_table()

        self.ax2.plot(t, v, "r-", linewidth=2)
        self.ax2.set_xlabel("Tiempo (s)")
        self.ax2.set_ylabel("Velocidad (m/s)")
        self.ax2.set_title("Velocidad vs Tiempo")
        self.ax2.grid(True, alpha=0.3)
        
        # Marcar punto final según espoleta
        if self.fuze_type.get() == "PDA":
            self.ax1.plot(x[-1], z[-1], 'ro', markersize=8, label='Impacto (PDA)')
        else:  # TIA
            proj_name = self.current_projectile.get('name', '').upper()
            is_base_ejection = any(x in proj_name for x in ['ILLUM', 'SMOKE', 'RAAM', 'M314', 'M485', 'HC'])
            
            if is_base_ejection:
                # Marcar punto de expulsión (si existe)
                if hasattr(self, 'tia_expulsion_point') and self.tia_expulsion_point:
                    x_exp, y_exp, z_exp = self.tia_expulsion_point
                    r_exp = np.sqrt(x_exp**2 + y_exp**2)
                    self.ax1.plot(r_exp, -z_exp, 'y*', markersize=12, 
                                  label=f'Expulsión TIA ({self.fuze_time.get():.1f}s)')
                # Marcar impacto final
                self.ax1.plot(x[-1], z[-1], 'ro', markersize=8, label='Impacto final')
            else:
                # HE con TIA: detonación en aire
                self.ax1.plot(x[-1], z[-1], 'y*', markersize=12, 
                              label=f'Detonación TIA ({t[-1]:.1f}s)')
        
        self.ax1.legend()
        
        # Punto arrastrable
        self.draggable_circle = self.ax1.plot(x[len(x)//2], z[len(z)//2], 'yo', markersize=10)[0]
        self.cursor_annotation = None
        
        self.fig.tight_layout()
        self.canvas.draw()
    

    def on_press(self, event):
        """Maneja el evento de presionar el mouse"""
        if event.inaxes == self.ax1 and self.trajectory_data is not None:
            self.drag_active = True
            self.update_circle_position(event)
    

    def on_release(self, event):
        """Maneja el evento de soltar el mouse"""
        self.drag_active = False
    

    def on_motion(self, event):
        """Maneja el evento de mover el mouse"""
        if hasattr(self, 'drag_active') and self.drag_active:
            if event.inaxes == self.ax1 and event.xdata is not None:
                self.update_circle_position(event)
    

    def update_circle_position(self, event):
        """Actualiza la posición del círculo arrastrable"""
        data = self.trajectory_data
        y = data["y"]
        t = data["t"]
        x = np.sqrt(y[0, :]**2 + y[1, :]**2)
        z = -y[2, :]
        v = np.sqrt(y[3, :]**2 + y[4, :]**2 + y[5, :]**2)
        
        idx = np.argmin(np.abs(x - event.xdata))
        
        self.draggable_circle.set_data([x[idx]], [z[idx]])
        
        if self.cursor_annotation is not None:
            self.cursor_annotation.remove()
            self.cursor_annotation = None
        
        mach_val = v[idx] / self.atmosphere.get_properties(z[idx])['speed_of_sound']
        
        x_max = np.max(x)
        z_max = np.max(z)
        
        if x[idx] < x_max * 0.5:
            x_offset = 40
        else:
            x_offset = -120
            
        if z[idx] < z_max * 0.5:
            y_offset = 40
        else:
            y_offset = -60
        
        self.cursor_annotation = self.ax1.annotate(
            f'Tiempo: {t[idx]:.2f} s\nAlcance: {x[idx]:.1f} m\nAltitud: {z[idx]:.1f} m\nVelocidad: {v[idx]:.1f} m/s\nMach: {mach_val:.3f}',
            xy=(x[idx], z[idx]),
            xytext=(x_offset, y_offset),
            textcoords='offset points',
            bbox=dict(boxstyle='round,pad=0.5', fc='yellow', alpha=0.9),
            arrowprops=dict(arrowstyle='->')
        )
        self.canvas.draw_idle()
    

    def update_table(self):
        """Actualiza la tabla de trayectoria"""
        for item in self.tree.get_children():
            self.tree.delete(item)
        
        data = self.trajectory_data
        y = data["y"]
        t = data["t"]
        
        range_vals = np.sqrt(y[0, :]**2 + y[1, :]**2)
        impact_range = range_vals[-1]
        deflection = y[1, :]
        alt = -y[2, :]
        speed = np.sqrt(y[3, :]**2 + y[4, :]**2 + y[5, :]**2)
        
        step = max(1, len(t) // 30)
        for i in range(0, len(t), step):
            mach = speed[i] / self.atmosphere.get_properties(alt[i])['speed_of_sound']
            self.tree.insert('', tk.END, values=(
                f"{t[i]:.2f}",
                f"{range_vals[i]:.1f}",
                f"{impact_range:.1f}",
                f"{deflection[i]:.2f}",
                f"{alt[i]:.1f}",
                f"{speed[i]:.1f}",
                f"{mach:.3f}"
            ))
        
        mach_impact = speed[-1] / self.atmosphere.get_properties(alt[-1])['speed_of_sound']
        self.tree.insert('', tk.END, values=(
            f"{t[-1]:.2f}",
            f"{range_vals[-1]:.1f}",
            f"{impact_range:.1f}",
            f"{deflection[-1]:.2f}",
            f"{alt[-1]:.1f}",
            f"{speed[-1]:.1f}",
            f"{mach_impact:.3f}"
        ), tags=('impact',))
        self.tree.tag_configure('impact', background='#ffcccc')

    def audit_query(self):
        """Procesa consultas de auditoría usando agentes inteligentes"""
        if self.trajectory_data is None:
            messagebox.showwarning("Sin datos", "Ejecuta una simulación primero")
            return
        
        query = self.audit_entry.get().lower()
        data = self.trajectory_data
        y = data["y"]
        t = data["t"]
        
        range_vals = np.sqrt(y[0, :]**2 + y[1, :]**2)
        alt = -y[2, :]
        speed = np.sqrt(y[3, :]**2 + y[4, :]**2 + y[5, :]**2)
        v0 = speed[0]
        elev = self.elevation.get()
        az = self.azimuth.get()
        mass = self.current_projectile["mass"]
        bc = self.current_projectile["ballistic_coefficient"]
        drag_func = self.current_projectile["drag_function"]
        final_deflection = y[1, -1]
        
        agent = self._determine_agent(query)
        
        response = "=" * 80 + "\n"
        response += f"  CONSULTA: {self.audit_entry.get()}\n"
        response += "=" * 80 + "\n\n"
        
        if agent['id'] == 'validation':
            response += self._agent_validation(query, v0, elev, az, mass, bc, drag_func)
        elif agent['id'] == 'convergence':
            response += self._agent_convergence(query, t, len(t), None)
        elif agent['id'] == 'empirical':
            response += self._agent_empirical(query, range_vals[-1], t[-1], final_deflection, drag_func, bc)
        else:
            response += self._agent_report(query, range_vals[-1], t[-1], np.max(alt), speed[-1], final_deflection)
        
        self.audit_text.config(state="normal")
        self.audit_text.delete(1.0, tk.END)
        self.audit_text.insert(1.0, response)
        self.audit_text.config(state="disabled")
    

    def _determine_agent(self, query):
        """Determina qué agente debe responder según palabras clave"""
        agents = [
            {
                'id': 'empirical',
                'keywords': ['deflexión', 'deflexion', 'deriva', 'coriolis', 'por qué', 'por que',
                            'comparar', 'real', 'empírico', 'empirico', 'tabla', 'nato', 'militar',
                            'diferencia', 'esperado', 'teórico', 'teorico', 'g1', 'g7', 'arrastre']
            },
            {
                'id': 'validation',
                'keywords': ['parámetro', 'parametro', 'masa', 'velocidad inicial', 'ángulo', 'angulo', 
                            'elevación', 'elevacion', 'balístico', 'balistico', 'carga', 'calibre',
                            'modificar', 'cambiar', 'aumentar', 'reducir', 'valor', 'rango', 'válido']
            },
            {
                'id': 'convergence',
                'keywords': ['precisión', 'precision', 'error', 'tolerancia', 'numérico', 'numerico',
                            'cálculo', 'calculo', 'calcula', 'integración', 'integracion', 'dop853',
                            'paso', 'convergencia', 'método', 'metodo']
            },
            {
                'id': 'report',
                'keywords': ['reporte', 'informe', 'exportar', 'resumen', 'todos', 'general']
            }
        ]
        
        query_lower = query.lower()
        for agent in agents:
            for keyword in agent['keywords']:
                if keyword in query_lower:
                    return agent
        
        return agents[0]
    

    def _agent_validation(self, query, v0, elev, az, mass, bc, drag_func):
        """Agente de Validación de Parámetros"""
        resp = "ANÁLISIS DE PARÁMETROS\n"
        resp += "-" * 80 + "\n\n"
        
        resp += "VERIFICACIÓN DE CONSISTENCIA:\n"
        resp += f"  • Velocidad inicial: {v0:.1f} m/s ✓\n"
        resp += f"  • Masa: {mass:.2f} kg ✓\n"
        resp += f"  • Coef. Balístico: {bc:.3f} ({drag_func}) ✓\n"
        resp += f"  • Elevación: {elev:.1f}° ✓\n\n"
        
        resp += "RANGOS DE VALIDEZ:\n"
        if 100 < v0 < 2000:
            resp += f"  • Velocidad: Dentro de rango balístico [100-2000] m/s\n"
        else:
            resp += f"  • Velocidad: ⚠️ Fuera de rango típico\n"
        
        if drag_func == 'G1':
            bc_range = (0.1, 1.0)
        else:
            bc_range = (0.3, 1.5)
        
        if bc_range[0] < bc < bc_range[1]:
            resp += f"  • BC {bc:.3f}: Dentro de rango para {drag_func} {bc_range}\n"
        else:
            resp += f"  • BC {bc:.3f}: ⚠️ Fuera de rango típico para {drag_func}\n"

        resp += f"  • Para MAYOR alcance: Aumente velocidad inicial (actual: {v0:.0f} m/s) o use G7\n"
        resp += f"  • Para MAYOR tiempo de vuelo: Aumente elevación (actual: {elev:.1f}°)\n"
        resp += f"  • Para MENOR deflexión: Reduzca latitud o tiempo de vuelo\n"
        
        return resp

    def _agent_empirical(self, query, range_val, time_val, deflection, drag_func, bc):
        """Agente de Correspondencia Empírica"""
        resp = "EXPLICACIÓN DETALLADA DEL CÁLCULO\n"
        resp += "=" * 80 + "\n\n"
        
        if self.reference_data is not None:
            resp += "COMPARACIÓN CON DATOS DE REFERENCIA:\n"
            resp += "-" * 80 + "\n"
            
            time_col = None
            for col in ['TOF', 'tof', 'tiempo', 'time', 'Tiempo', 'Time']:
                if col in self.reference_data.columns:
                    time_col = col
                    break
            
            range_col = None
            for col in ['DIST', 'dist', 'alcance', 'range', 'Alcance', 'Range']:
                if col in self.reference_data.columns:
                    range_col = col
                    break
            
            drift_col = None
            for col in ['DRIFT', 'drift', 'deflexion', 'deflection', 'deriva']:
                if col in self.reference_data.columns:
                    drift_col = col
                    break
            
            if range_col:
                ref_ranges = self.reference_data[range_col].values
                closest_idx = np.argmin(np.abs(ref_ranges - range_val))
                ref_range = ref_ranges[closest_idx]
                
                resp += f"\nCOMPARACIÓN PARA ALCANCE ~{ref_range:.0f} m:\n"
                resp += f"  (Simulado: {range_val:.0f} m)\n"
                resp += "-" * 40 + "\n"
                
                if time_col:
                    ref_time = self.reference_data[time_col].iloc[closest_idx]
                    diff = time_val - ref_time
                    pct = 100 * diff / ref_time if ref_time != 0 else 0
                    resp += f"\nTIEMPO DE VUELO:\n"
                    resp += f"  • Simulado: {time_val:.2f} s\n"
                    resp += f"  • Referencia (TOF): {ref_time:.2f} s\n"
                    resp += f"  • Diferencia: {diff:+.2f} s ({pct:+.1f}%)\n"
                    
                    if abs(pct) > 2:
                        resp += f"\n  PARA LLEGAR A {ref_time:.2f} s DEBES:\n"
                        if diff > 0:
                            resp += f"    - Reducir velocidad inicial en ~{abs(pct)*2:.0f} m/s\n"
                            resp += f"    - O reducir elevación en ~{abs(pct):.1f} mils\n"
                        else:
                            resp += f"    - Aumentar velocidad inicial en ~{abs(pct)*2:.0f} m/s\n"
                            resp += f"    - O aumentar elevación en ~{abs(pct):.1f} mils\n"
                
                if drift_col:
                    ref_drift = self.reference_data[drift_col].iloc[closest_idx]
                    drift_meters = ref_drift * (ref_range / 1000)
                    diff_drift = deflection - drift_meters
                    pct_drift = 100 * diff_drift / drift_meters if drift_meters != 0 else 0
                    
                    resp += f"\nDEFLEXIÓN:\n"
                    resp += f"  • Simulada: {deflection:.2f} m\n"
                    resp += f"  • Referencia (DRIFT): {ref_drift:.2f} mils (~{drift_meters:.2f} m)\n"
                    resp += f"  • Diferencia: {diff_drift:+.2f} m\n"
                    
                    if abs(diff_drift) > 1:
                        resp += f"\n  POSIBLES CAUSAS:\n"
                        resp += f"    • Efecto Coriolis no coincide con tabla\n"
                        resp += f"    • Ajuste de latitud o azimuth necesario\n"
            
            return resp
        
        v0 = self.current_projectile['muzzle_velocity']
        elev = self.elevation.get()
        mass = self.current_projectile['mass']
        lat = self.latitude.get()
        
        if "deflexión" in query or "deflexion" in query:
            resp += "CÁLCULO DE LA DEFLEXIÓN (EFECTO CORIOLIS):\n"
            resp += "-" * 80 + "\n\n"
            resp += "OPERACIONES REALIZADAS:\n"
            resp += "  1. Velocidad angular terrestre: ω = 7.2921159×10⁻⁵ rad/s\n"
            resp += f"  2. Latitud: φ = {lat:.1f}°\n"
            resp += f"  3. Componentes de Coriolis:\n"
            resp += f"     • ω_N = ω × cos(φ) = {7.2921159e-5 * np.cos(np.radians(lat)):.6f} rad/s\n"
            resp += f"     • ω_D = -ω × sin(φ) = {-7.2921159e-5 * np.sin(np.radians(lat)):.6f} rad/s\n"
            resp += f"  4. Aceleración lateral: a_y = -2 × (ω_D × v_x - ω_N × v_z)\n"
            resp += f"  5. Integración numérica sobre {time_val:.2f} s\n\n"
            resp += "VALORES UTILIZADOS:\n"
            resp += f"  • Tiempo total: {time_val:.2f} s\n"
            resp += f"  • Velocidad promedio: {range_val/time_val:.1f} m/s\n"
            resp += f"  • Deflexión calculada: {deflection:.2f} m\n"
            resp += f"  • Tasa de deriva: {deflection/time_val:.2f} m/s\n\n"
            resp += "CÓMO MODIFICAR LA DEFLEXIÓN:\n"
            resp += f"  • Para REDUCIR a X metros: Multiplica X/{deflection:.2f} por el tiempo de vuelo\n"
            resp += f"  • Para CERO deflexión: Apunte {deflection/time_val*1000:.1f} mils a la IZQUIERDA\n"
            
        elif "alcance" in query:
            resp += "CÁLCULO DEL ALCANCE:\n"
            resp += "-" * 80 + "\n\n"
            resp += "OPERACIONES REALIZADAS:\n"
            resp += "  1. Ecuaciones de movimiento: dv/dt = -D(v,ρ,BC) - g + Coriolis, dx/dt = v\n"
            resp += f"  2. Integración DOP853 con {len(self.trajectory_data['t'])} pasos\n"
            resp += f"  3. Alcance = √(X_final² + Y_final²)\n\n"
            resp += "VALORES UTILIZADOS:\n"
            resp += f"  • Velocidad inicial: {v0:.1f} m/s\n"

            resp += f"  • Densidad del aire (nivel mar): 1.225 kg/m³\n\n"

    def audit_calculate_how_to_reach(self):
        """Calcula cómo llegar a valores objetivo"""
        if self.trajectory_data is None:
            messagebox.showwarning("Sin datos", "Ejecuta una simulación primero")
            return
        
        data = self.trajectory_data
        y = data["y"]
        t = data["t"]
        current_range = np.sqrt(y[0, -1]**2 + y[1, -1]**2)
        current_time = t[-1]
        current_defl = y[1, -1]
        
        v0 = self.current_projectile["muzzle_velocity"]
        elev = self.elevation.get()
        lat = self.latitude.get()
        
        result_text = "CÁLCULOS PRECISOS PARA LLEGAR AL OBJETIVO\n"
        result_text += "=" * 50 + "\n\n"
        result_text += "FÓRMULAS UTILIZADAS (ARES/USSA 1976/WGS84):\n"
        result_text += "-" * 50 + "\n"
        
        if self.target_range.get():
            target = float(self.target_range.get())
            factor = target / current_range
            
            result_text += f"\nALCANCE OBJETIVO: {target:.0f} m\n"
            result_text += f"  Actual: {current_range:.2f} m\n"
            result_text += f"  Factor: {factor:.4f}\n\n"
            
            v0_new = v0 * (factor ** (1/1.8))
            result_text += f"  Nueva velocidad ≈ {v0_new:.0f} m/s\n"
            
            sin_2theta = np.sin(2 * np.radians(elev))
            sin_2theta_new = sin_2theta * (factor ** (1/0.7))
            sin_2theta_new = np.clip(sin_2theta_new, -1, 1)
            elev_new = np.degrees(np.arcsin(sin_2theta_new)) / 2
            result_text += f"  O nueva elevación ≈ {elev_new:.1f}°\n"
            
            coriolis_effect = 2 * 7.292e-5 * v0 * np.sin(np.radians(lat)) * current_time
            result_text += f"  Efecto Coriolis en alcance: {coriolis_effect:.1f} m\n"
        
        if self.target_time.get():
            target = float(self.target_time.get())
            factor = target / current_time
            
            result_text += f"\nTIEMPO OBJETIVO: {target:.2f} s\n"
            result_text += f"  Actual: {current_time:.2f} s\n"
            result_text += f"  Factor: {factor:.4f}\n\n"
            
            elev_new = np.degrees(np.arcsin(np.sin(np.radians(elev)) * (factor ** (1/0.6))))
            elev_new = np.clip(elev_new, 0, 90)
            result_text += f"  Nueva elevación ≈ {elev_new:.1f}°\n"
            
            v0_new = v0 * (factor ** (1/0.9))
            result_text += f"  O nueva velocidad ≈ {v0_new:.0f} m/s\n"
        
        if self.target_deflection.get():
            target = float(self.target_deflection.get())
            diff = target - current_defl
            
            result_text += f"\nDEFLEXIÓN OBJETIVO: {target:.2f} m\n"
            result_text += f"  Actual: {current_defl:.2f} m\n"
            result_text += f"  Diferencia: {diff:+.2f} m\n\n"
            
            omega = 7.2921159e-5
            sin_lat = np.sin(np.radians(lat))
            v_avg = current_range / current_time
            coriolis_per_second = 2 * omega * sin_lat * v_avg
            result_text += f"  Tasa de deriva Coriolis: {coriolis_per_second:.4f} m/s\n"
            
            if abs(diff) > 0.01:
                correction_rad = np.arcsin(diff / (current_defl + 1e-10))
                correction_mils = np.degrees(correction_rad) * 17.777
                result_text += f"  Corrección azimuth: {correction_mils:+.1f} mils\n"
                v0_adjustment = diff / (coriolis_per_second * current_time)
                result_text += f"  O ajuste de velocidad: {v0_adjustment:+.1f} m/s\n"
        
        result_text += "\n" + "=" * 50 + "\n"
        result_text += "NOTA: Use el panel superior para aplicar los cambios.\n"
        
        self.audit_what_to_change.config(state="normal")
        self.audit_what_to_change.delete(1.0, tk.END)
        self.audit_what_to_change.insert(1.0, result_text)
        self.audit_what_to_change.config(state="disabled")

    def export_csv(self):
        """Exporta los datos de trayectoria a CSV"""
        if self.trajectory_data is None:
            messagebox.showwarning("Sin datos", "Ejecuta una simulación primero")
            return
        
        filename = filedialog.asksaveasfilename(defaultextension=".csv",
            filetypes=[("CSV files", "*.csv"), ("All files", "*.*")])
        
        if filename:
            data = self.trajectory_data
            y = data["y"]
            import pandas as pd
            
            range_vals = np.sqrt(y[0, :]**2 + y[1, :]**2)
            alt = -y[2, :]
            speed = np.sqrt(y[3, :]**2 + y[4, :]**2 + y[5, :]**2)
            mach = [speed[i] / self.atmosphere.get_properties(alt[i])['speed_of_sound'] for i in range(len(data["t"]))]
            
            df = pd.DataFrame({
                "time": data["t"], "range": range_vals, "deflection": y[1, :],
                "x": y[0, :], "y": y[1, :], "z": y[2, :],
                "vx": y[3, :], "vy": y[4, :], "vz": y[5, :],
                "altitude": alt, "speed": speed, "mach": mach
            })
            df.to_csv(filename, index=False)
            messagebox.showinfo("Éxito", f"Datos exportados a {filename}")

    def update_3d_plot(self):
        """Actualiza el gráfico 3D con la trayectoria y puntos de impacto."""
        if self.trajectory_data is None:
            return
        
        self.ax3d.clear()
        
        data = self.trajectory_data
        y = data["y"]
        t = data["t"]
        
        # Trayectoria original
        x_orig = y[0, :]
        y_orig = y[1, :]
        z_orig = -y[2, :]
        
        # Plotear trayectoria con gradiente de color
        colors = plt.cm.viridis(np.linspace(0, 1, len(x_orig)))
        for i in range(len(x_orig)-1):
            self.ax3d.plot(x_orig[i:i+2], y_orig[i:i+2], z_orig[i:i+2], 
                          color=colors[i], linewidth=2.5, alpha=0.8)
        
        # Punto de origen (pieza)
        self.ax3d.scatter([0], [0], [0], c='green', s=150, marker='^', 
                         label='🎯 Origen', alpha=0.9, edgecolors='darkgreen', linewidth=2)
        
        # Objetivo
        target_range = np.sqrt(x_orig[-1]**2 + y_orig[-1]**2)
        target_x = target_range * np.cos(np.radians(self.azimuth.get()))
        target_y = target_range * np.sin(np.radians(self.azimuth.get()))
        self.ax3d.scatter([target_x], [target_y], [0], c='blue', s=200, marker='*', 
                         label='🎯 Objetivo', alpha=1.0, edgecolors='darkblue', linewidth=2)
        
        # Punto de impacto original
        self.ax3d.scatter([x_orig[-1]], [y_orig[-1]], [z_orig[-1]], 
                         c='red', s=120, marker='o', label='💥 Impacto Original', 
                         alpha=0.8, edgecolors='darkred', linewidth=2)
        
        # Calcular impacto corregido
        elev_corr = self.correction_elevation.get()
        az_corr = self.correction_azimuth.get()
        
        elev_corr_deg = elev_corr * 360 / 6400
        az_corr_deg = az_corr * 360 / 6400
        
        impact_vec = np.array([x_orig[-1], y_orig[-1]])
        impact_dist = np.linalg.norm(impact_vec)
        
        new_az = self.azimuth.get() + az_corr_deg
        corrected_x = impact_dist * np.cos(np.radians(new_az))
        corrected_y = impact_dist * np.sin(np.radians(new_az))
        
        # Punto de impacto corregido
        self.ax3d.scatter([corrected_x], [corrected_y], [z_orig[-1]], 
                         c='orange', s=140, marker='s', label='✨ Impacto Corregido', 
                         alpha=0.9, edgecolors='darkorange', linewidth=2)
        
        # Línea de error original
        self.ax3d.plot([x_orig[-1], target_x], [y_orig[-1], target_y], [0, 0], 
                      'r--', linewidth=1.5, alpha=0.6, label='Error Original')
        
        # Línea de error corregido
        self.ax3d.plot([corrected_x, target_x], [corrected_y, target_y], [0, 0], 
                      'g--', linewidth=1.5, alpha=0.6, label='Error Corregido')
        
        # Calcular errores
        error_vec = np.array([target_x - x_orig[-1], target_y - y_orig[-1]])
        error_dist = np.linalg.norm(error_vec)
        
        corrected_error_vec = np.array([target_x - corrected_x, target_y - corrected_y])
        corrected_error_dist = np.linalg.norm(corrected_error_vec)
        
        # Actualizar label de error con color según precisión
        if corrected_error_dist < 10:
            error_color = "green"
            precision = "¡EXCELENTE!"
        elif corrected_error_dist < 50:
            error_color = "orange"
            precision = "BUENA"
        else:
            error_color = "red"
            precision = "CORREGIR"
        
        self.sim3d_error_label.config(
            text=f"Error: {error_dist:.1f} m → {corrected_error_dist:.1f} m ({precision})",
            foreground=error_color
        )
        
        # Actualizar barra de progreso (máximo error = 100m)
        progress_value = max(0, 100 - min(corrected_error_dist, 100))
        self.error_progress['value'] = progress_value
        self.error_progress['style'] = 'green.Horizontal.TProgressbar' if progress_value > 90 else 'orange.Horizontal.TProgressbar'
        
        # Detalles del error
        error_text = f"Error Original:  {error_dist:.2f} m\n"
        error_text += f"Error Corregido: {corrected_error_dist:.2f} m\n"
        error_text += f"Mejora:          {error_dist - corrected_error_dist:+.2f} m\n"
        error_text += f"Deflexión orig:  {y_orig[-1]:.2f} m"
        
        self.sim3d_error_details.config(state="normal")
        self.sim3d_error_details.delete(1.0, tk.END)
        self.sim3d_error_details.insert(1.0, error_text)
        self.sim3d_error_details.config(state="disabled")
        
        # Configurar ejes
        self.ax3d.set_xlabel('Alcance X (m)', fontsize=11, fontweight='bold')
        self.ax3d.set_ylabel('Deflexión Y (m)', fontsize=11, fontweight='bold')
        self.ax3d.set_zlabel('Altitud Z (m)', fontsize=11, fontweight='bold')
        self.ax3d.set_title('🎯 Simulador 3D de Corrección de Tiro', fontsize=13, fontweight='bold', pad=20)
        
        # Ajustar límites
        max_range = max(x_orig[-1], target_x) * 1.15
        self.ax3d.set_xlim([-max_range*0.05, max_range])
        self.ax3d.set_ylim([-max_range*0.15, max_range*0.15])
        self.ax3d.set_zlim([0, max(z_orig)*1.3])
        
        # Leyenda y grid
        self.ax3d.legend(loc='upper left', fontsize=9, framealpha=0.9)
        self.ax3d.grid(True, alpha=0.2)
        
        # Vista inicial mejorada
        self.ax3d.view_init(elev=30, azim=-45)
        
        self.canvas3d.draw()
        self.sim3d_status.config(text=f"✅ Corrección aplicada: Elev {elev_corr:+.1f} mils | Az {az_corr:+.1f} mils")

    def search_firing_solutions(self):
        """Busca todas las combinaciones proyectil/carga/arrastre que alcanzan el objetivo."""
        try:
            target_distance = float(self.target_distance.get())
            target_azimuth = float(self.target_azimuth.get())
            max_tof = float(self.max_tof.get()) if self.max_tof.get() else 999
            for item in self.solutions_tree.get_children():
                self.solutions_tree.delete(item)
            solutions = []
            for proj_file in os.listdir("data/projectiles"):
                if not proj_file.endswith(".json"):
                    continue
                with open(f"data/projectiles/{proj_file}", "r", encoding="utf-8") as f:
                    proj = json.load(f)
                caliber = proj.get("caliber", 0)
                if caliber == 105 and not self.include_105mm.get():
                    continue
                if caliber == 155 and not self.include_155mm.get():
                    continue
                for charge_name, charge_data in proj.get("charges", {}).items():
                    v0 = charge_data["muzzle_velocity"]
                    for drag_model in ["G1", "G7"]:
                        temp_proj = proj.copy()
                        temp_proj["muzzle_velocity"] = v0
                        temp_proj["drag_function"] = drag_model
                        generator = FiringTableGenerator(temp_proj, self.atmosphere, latitude=self.latitude.get(), wind_speed=self.wind_speed.get(), wind_direction=self.wind_direction.get())
                        r_max = generator.simulate_trajectory(45.0, target_azimuth)["range"]
                        if target_distance > r_max:
                            continue
                        if self.tiro_tenso_only.get():
                            min_e, max_e = 0.5, 45.0
                        elif self.tiro_curvo_only.get():
                            min_e, max_e = 45.0, 85.0
                        else:
                            min_e, max_e = 0.5, 85.0
                        elev = generator.find_elevation_for_range(target_distance, target_azimuth, min_elev=min_e, max_elev=max_e)
                        if elev is None:
                            continue
                        result = generator.simulate_trajectory(elev, target_azimuth)
                        if result["time"] > max_tof:
                            continue
                        qe_mils = elev * (6400 / 360)
                        solutions.append({"proj_name": proj["name"], "charge": charge_name, "drag": drag_model, "qe": qe_mils, "elev": elev, "tof": result["time"], "max_alt": result["max_alt"], "v0": v0, "proj_file": f"data/projectiles/{proj_file}"})
            solutions.sort(key=lambda x: x["qe"])
            for sol in solutions:
                self.solutions_tree.insert("", "end", values=(sol["proj_name"], sol["charge"], sol["drag"], f"{sol["qe"]:.1f}", f"{sol["tof"]:.1f}", f"{sol["max_alt"]:.1f}"))
            if solutions:
                best = solutions[0]
                self.recommendation_label.config(text=f"SOLUCIÓN RECOMENDADA: {best["proj_name"]} + {best["charge"]} + {best["drag"]} | QE={best["qe"]:.1f} mils ({best["elev"]:.2f}°) | TOF={best["tof"]:.1f}s")
                self.best_solution = best
            else:
                self.recommendation_label.config(text="No se encontraron soluciones con las restricciones dadas.")
                self.best_solution = None
        except Exception as e:
            messagebox.showerror("Error", f"Error al buscar soluciones: {str(e)}")

    def apply_selected_solution(self):
        selection = self.solutions_tree.selection()
        if not selection:

        values = self.solutions_tree.item(selection[0])["values"]
        proj_name = values[0]

        self.drag_function.set(drag)
        self.on_drag_changed(None)
        self.elevation.set(elevation)
        self.notebook.select(0)
        messagebox.showinfo("Solución Aplicada", f"Proyectil: {proj_name}\nCarga: {charge}\nArrastre: {drag}\nElevación: {elevation:.2f}° ({qe_mils:.1f} mils)")

    def generate_firing_table(self):
        """Genera la tabla de tiro."""
        if not self.current_projectile:
            messagebox.showerror("Error", "No hay proyectil cargado")
            return
        
        try:
            start = int(self.table_start_range.get())
            end = int(self.table_end_range.get())
            step = int(self.table_step.get())
            azimuth = float(self.table_azimuth.get())
            
            from src.tables.firing_table_generator import FiringTableGenerator
            
            generator = FiringTableGenerator(
                self.current_projectile,
                self.atmosphere,
                latitude=self.latitude.get(),
                wind_speed=self.wind_speed.get(),
                wind_direction=self.wind_direction.get(),
                use_magnus=self.use_magnus.get(),
                spin_rate_rpm=self.spin_rate_rpm.get(),
                rifling_twist=self.rifling_twist.get()
            )

# 1. ENTORNO BALISTICO (ISA)
# ==================================================================
class EntornoBalistico:
    def __init__(self, latitud_grados=14.08, viento_x=0.0, viento_z=0.0, temperatura_polvora=21.0):
        self.latitud = math.radians(latitud_grados)
        self.omega_tierra = 7.2921159e-5
        self.radio_tierra = 6371000.0
        self.g0 = 9.80665
        self.temp_0 = 288.15
        self.presion_0 = 101325.0
        self.L = 0.0065
        self.R = 287.05
        self.viento_x = viento_x
        self.viento_z = viento_z
        self.temp_polvora = temperatura_polvora

    def get_gravedad(self, altitud):
        return self.g0 * (self.radio_tierra / (self.radio_tierra + altitud)) ** 2

    def atmosfera(self, altitud):
        if altitud < 11000:
            T = self.temp_0 - self.L * altitud
            P = self.presion_0 * (T / self.temp_0) ** (self.g0 / (self.L * self.R))
        else:
            T_11 = self.temp_0 - self.L * 11000
            P_11 = self.presion_0 * (T_11 / self.temp_0) ** (self.g0 / (self.L * self.R))
            T = T_11
            P = P_11 * math.exp(-self.g0 * (altitud - 11000) / (self.R * T))
        rho = P / (self.R * T)
        return rho, T

    def correccion_v0_por_temperatura(self, v0_nominal):
        delta = 0.0008 * (self.temp_polvora - 21.0)
        return v0_nominal * (1.0 + delta)

# 2. AERODINAMICA - Tabla G7
# ==================================================================
class CalculadoraAerodinamica:
    TABLA_G7 = [
        (0.00, 0.120), (0.70, 0.122), (0.85, 0.135), (0.95, 0.200),
        (1.00, 0.350), (1.05, 0.380), (1.10, 0.375), (1.20, 0.355),
        (1.50, 0.310), (2.00, 0.260), (3.00, 0.190), (5.00, 0.150)
    ]

    @staticmethod
    def cd_g7(mach):
        tabla = CalculadoraAerodinamica.TABLA_G7
        if mach <= tabla[0][0]:
            return tabla[0][1]
        if mach >= tabla[-1][0]:
            return tabla[-1][1]
        for i in range(len(tabla)-1):
            if tabla[i][0] <= mach <= tabla[i+1][0]:
                m1, c1 = tabla[i]
                m2, c2 = tabla[i+1]
                return c1 + (c2 - c1) * (mach - m1) / (m2 - m1)
        return 0.2

# 3. DEFINICION DE PROYECTILES
# ==================================================================
class Proyectil:
    def __init__(self, nombre, calibre_m, masa_kg, v0_nominal_mps, factor_forma=1.0,
                 es_hera=False, empuje_N=0, t_empuje_inicio=0, t_empuje_fin=0, 
                 perdida_masa_kg_s=0.0, fuente=""):
        self.nombre = nombre
        self.calibre = calibre_m
        self.masa = masa_kg
        self.v0_nominal = v0_nominal_mps
        self.factor_forma = factor_forma
        self.es_hera = es_hera
        self.empuje_N = empuje_N
        self.t_empuje_inicio = t_empuje_inicio
        self.t_empuje_fin = t_empuje_fin
        self.perdida_masa_kg_s = perdida_masa_kg_s
        self.fuente = fuente

    def cd(self, mach):
        return CalculadoraAerodinamica.cd_g7(mach) * self.factor_forma

    def masa_actual(self, t):
        if not self.es_hera or t < self.t_empuje_inicio:
            return self.masa
        if t > self.t_empuje_fin:
            return self.masa - self.perdida_masa_kg_s * (self.t_empuje_fin - self.t_empuje_inicio)
        return self.masa - self.perdida_masa_kg_s * (t - self.t_empuje_inicio)

# 4. MOTOR BALISTICO (RK4)
# ==================================================================
class MotorBalistico:
    def __init__(self, entorno=None):
        self.env = entorno if entorno else EntornoBalistico()

    def derivadas(self, t, estado, proy):
        x, y, z, vx, vy, vz = estado
        rho, temp_k = self.env.atmosfera(y)
        g = self.env.get_gravedad(y)

        v_air_x = vx - self.env.viento_x
        v_air_y = vy
        v_air_z = vz - self.env.viento_z
        v_total_air = math.hypot(v_air_x, v_air_y, v_air_z)
        if v_total_air < 0.001:
            v_total_air = 0.001

        c_sonido = math.sqrt(1.4 * self.env.R * temp_k)
        mach = v_total_air / c_sonido
        cd = proy.cd(mach)
        area = math.pi * (proy.calibre/2)**2
        arrastre = 0.5 * rho * v_total_air**2 * cd * area

        empuje = 0.0
        masa_actual = proy.masa_actual(t)
        if proy.es_hera and proy.t_empuje_inicio <= t <= proy.t_empuje_fin:
            empuje = proy.empuje_N

        ax = (empuje * (vx/v_total_air) - arrastre * (v_air_x/v_total_air)) / masa_actual
        ay = -g + (empuje * (vy/v_total_air) - arrastre * (v_air_y/v_total_air)) / masa_actual

        if proy.calibre > 0.13:
            k_spin = 1.8e-5
            factor_cuadratico = 1.0 + 0.02 * (max(0, x) / 1000.0)
        else:
            k_spin = 2.5e-5
            factor_cuadratico = 1.0 + 0.015 * (max(0, x) / 1000.0)
        a_spin = k_spin * vx * (max(0, x) / 1000.0) * factor_cuadratico

        a_coriolis = -2.0 * self.env.omega_tierra * vx * math.sin(self.env.latitud)
        az = (-arrastre * (v_air_z/v_total_air) / masa_actual) + a_coriolis + a_spin

        return [vx, vy, vz, ax, ay, az]

    def simular_rk4(self, proy, qe_mils, dt=0.1, tiempo_max=120.0):
        v0_corregida = self.env.correccion_v0_por_temperatura(proy.v0_nominal)
        qe_rad = qe_mils * math.pi / 3200.0
        vx0 = v0_corregida * math.cos(qe_rad)
        vy0 = v0_corregida * math.sin(qe_rad)
        estado = [0.0, 0.0, 0.0, vx0, vy0, 0.0]
        t = 0.0
        y_max = 0.0

        while t < tiempo_max:
            caida = (estado[0]**2) / (2.0 * self.env.radio_tierra)
            if estado[1] < -caida and t > 0.5:
                break

            k1 = self.derivadas(t, estado, proy)
            e2 = [estado[i] + 0.5*dt*k1[i] for i in range(6)]
            k2 = self.derivadas(t+0.5*dt, e2, proy)
            e3 = [estado[i] + 0.5*dt*k2[i] for i in range(6)]
            k3 = self.derivadas(t+0.5*dt, e3, proy)
            e4 = [estado[i] + dt*k3[i] for i in range(6)]
            k4 = self.derivadas(t+dt, e4, proy)

            for i in range(6):
                estado[i] += (dt/6.0)*(k1[i] + 2*k2[i] + 2*k3[i] + k4[i])

            t += dt
            if estado[1] > y_max:
                y_max = estado[1]

        return estado[0], estado[2], t, y_max

# 5. GENERADOR DE TABLAS FT (FORMATO ESTANDAR MILITAR)
# ==================================================================
class GeneradorTablasFT:
    def __init__(self, motor):
        self.motor = motor
        self.KNOT_TO_MPS = 0.514444

    def encontrar_qe_para_alcance(self, proy, alcance_objetivo, qe_min, qe_max, tolerancia=10):
        for _ in range(30):
            qe = (qe_min + qe_max) / 2.0
            alcance, _, _, _ = self.motor.simular_rk4(proy, qe)
            if abs(alcance - alcance_objetivo) < tolerancia:
                return qe
            if alcance < alcance_objetivo:
                qe_min = qe
            else:
                qe_max = qe
        return (qe_min + qe_max) / 2.0

    def generar_tabla_ft(self, proy, qe_inicio, qe_fin, qe_step, verbose=True):
        """
        Genera tabla de tiro en formato FT estándar
        Columnas:
        1. QE (MILS)
        2. Corrección QE proyectil alternativo (MILS) - 0 para base
        3. Corrección por altura +50m (MILS) - estimado
        4. Corrección por rango +100m (MILS) - estimado
        5. Corrección por viento 1 nudo (METROS) - estimado
        6. Tiempo de vuelo (SEG)
        7. Alcance (METROS)
        8. Corrección deflexión (MILS)
        """
        resultados = []

        if verbose:
            print("\n" + "="*110)
            print(f"TABLA DE TIRO - {proy.nombre}")
            print(f"v0 = {proy.v0_nominal} m/s | Masa = {proy.masa} kg | Fuente: {proy.fuente}")
            print("="*110)
            print(f"{'QE':>6} {'CORR_PRJ':>10} {'CORR_H50':>10} {'CORR_R100':>10} {'WIND_1KT':>10} {'TOF':>8} {'RANGE':>10} {'DEFL':>10}")
            print(f"{'(MILS)':>6} {'(MILS)':>10} {'(MILS)':>10} {'(MILS)':>10} {'(M)':>10} {'(SEC)':>8} {'(M)':>10} {'(MILS)':>10}")
            print("-"*110)

        # Primera pasada: calcular todos los alcances
        alcances_por_qe = {}
        qe = qe_inicio
        while qe <= qe_fin:
            alcance, deriva, tof, altura_max = self.motor.simular_rk4(proy, qe)
            if alcance > 100:
                alcances_por_qe[qe] = (alcance, deriva, tof)
            qe += qe_step

        # Segunda pasada: calcular correcciones
        qe_list = sorted(alcances_por_qe.keys())

        for i, qe in enumerate(qe_list):
            alcance, deriva, tof = alcances_por_qe[qe]

            # Corrección por altura (estimada basada en pendiente)
            if i > 0 and i < len(qe_list) - 1:
                qe_prev = qe_list[i-1]
                qe_next = qe_list[i+1]
                alc_prev = alcances_por_qe[qe_prev][0]
                alc_next = alcances_por_qe[qe_next][0]

                # d(QE)/d(alcance) * 50m = corrección aproximada
                dqe_dalc = (qe_next - qe_prev) / (alc_next - alc_prev) if (alc_next - alc_prev) > 0 else 0
                corr_altura = dqe_dalc * 50  # Corrección para +50m de altura

                # Corrección por rango +100m
                corr_rango = dqe_dalc * 100
            else:
                corr_altura = 0.5  # Valor por defecto
                corr_rango = 1.0

            # Corrección por viento (simulación rápida)
            self.motor.env.viento_x = self.KNOT_TO_MPS  # 1 nudo de frente
            alcance_viento, _, _, _ = self.motor.simular_rk4(proy, qe)
            self.motor.env.viento_x = 0
            corr_viento = alcance - alcance_viento  # Positivo = viento reduce alcance

            # Corrección de deflexión
            if alcance > 0:
                defl_mils = (deriva / alcance) * 1000
            else:
                defl_mils = 0

            resultados.append({
                'qe': round(qe, 1),
                'corr_proy': 0.0,  # Base
                'corr_altura': round(abs(corr_altura), 1),
                'corr_rango': round(abs(corr_rango), 1),
                'corr_viento': round(corr_viento, 1),
                'tof': round(tof, 1),
                'alcance': round(alcance, 0),
                'corr_defl': round(defl_mils, 1)
            })

            if verbose:
                print(f"{qe:>6.1f} {0.0:>10.1f} {abs(corr_altura):>10.1f} {abs(corr_rango):>10.1f} {corr_viento:>10.1f} {tof:>8.1f} {alcance:>10.0f} {defl_mils:>10.1f}")

        return resultados
