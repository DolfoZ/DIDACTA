  useEffect(() => {

import express from 'express';
import { Pool } from 'pg';
import { verifyToken, verifyRole } from '../middleware/auth';

const router = express.Router();
const pool = new Pool({ connectionString: process.env.DATABASE_URL });

// Obtener todos los batallones
router.get('/batallones', verifyToken, async (req, res) => {
  const result = await pool.query('SELECT * FROM batallones ORDER BY id');
  res.json(result.rows);
});

// Obtener personal por batallón
router.get('/batallones/:id/personal', verifyToken, async (req, res) => {
  const result = await pool.query(
    'SELECT * FROM personal WHERE batallon_id = $1 ORDER BY grado, nombre',
    [req.params.id]
  );
  res.json(result.rows);
});

// Inventario de municiones por batallón
router.get('/batallones/:id/municiones', verifyToken, async (req, res) => {
  const result = await pool.query(
    `SELECT m.*, 
      CASE WHEN m.cantidad_actual <= m.cantidad_minima THEN 'critico'
           WHEN m.cantidad_actual <= m.cantidad_minima * 2 THEN 'alerta'
           ELSE 'normal' END as nivel
     FROM municiones m WHERE m.batallon_id = $1`,
    [req.params.id]
  );
  res.json(result.rows);
});

// Estado de obuses
router.get('/batallones/:id/obuses', verifyToken, async (req, res) => {
  const result = await pool.query(
    'SELECT * FROM obuses WHERE batallon_id = $1 ORDER BY modelo',
    [req.params.id]
  );
  res.json(result.rows);
});

// Dashboard logístico consolidado
router.get('/dashboard', verifyToken, async (req, res) => {
  const [personal, obuses, municiones, misiones] = await Promise.all([
    pool.query('SELECT b.nombre, COUNT(p.id) as total FROM batallones b LEFT JOIN personal p ON p.batallon_id = b.id GROUP BY b.nombre'),
    pool.query("SELECT b.nombre, COUNT(o.id) as total, COUNT(o.id) FILTER (WHERE o.estado = 'operativo') as operativos FROM batallones b LEFT JOIN obuses o ON o.batallon_id = b.id GROUP BY b.nombre"),
    pool.query("SELECT b.nombre, SUM(m.cantidad_actual) as total_municion FROM batallones b LEFT JOIN municiones m ON m.batallon_id = b.id GROUP BY b.nombre"),
    pool.query("SELECT COUNT(*) FILTER (WHERE estado = 'pendiente') as pendientes, COUNT(*) FILTER (WHERE estado = 'ejecutando') as ejecutando FROM misiones_fuego")
  ]);
  
  res.json({
    personal: personal.rows,
    obuses: obuses.rows,
    municiones: municiones.rows,
    misiones: misiones.rows[0]
  });
});

export default router;

MUNITIONS_DIR = '/opt/ares/drev/data/projectiles'

class SimulationRequest(BaseModel):
    projectile_file: str  # ej: "M107.json"
    charge: str           # ej: "WB M4A2"
    elevation_deg: float
    azimuth_deg: float = 0.0
    latitude_deg: float = 45.0
    temp_offset: float = 0.0
    humidity: float = 0.5
    wind_speed: float = 0.0
    wind_direction: float = 0.0
    use_magnus: bool = False
    drag_function: str = "G1"

@app.post("/api/drev/simulate")
async def simulate_trajectory(req: SimulationRequest):
    """Ejecuta tu simulación balística real y devuelve resultados"""
    try:
        # Cargar proyectil desde tu JSON
        with open(os.path.join(MUNITIONS_DIR, req.projectile_file)) as f:
            proj = json.load(f)
        
        # Aplicar carga seleccionada
        charge_data = proj['charges'][req.charge]
        v0 = charge_data['muzzle_velocity']
        bc = proj['ballistic_coefficient']
        mass = proj['mass']
        caliber = proj['caliber'] / 1000.0  # convertir mm a m
        
        # Inicializar tus modelos
        atmosphere = MolecularAir()
        atmosphere.set_temperature_offset(req.temp_offset)
        atmosphere.set_humidity(req.humidity)
        
        drag = DragCoefficient(req.drag_function, bc)
        geom = ProjectileGeometry(proj['caliber'], proj['caliber_unit'])
        coriolis = Coriolis(req.latitude_deg)
        wind = WindModel(req.wind_speed, req.wind_direction)
        forces = Forces(drag, geom, mass, coriolis_model=coriolis, 
                       latitude=req.latitude_deg, wind_model=wind)
        
        # Tu lógica de simulación (RESPETANDO TU CÓDIGO ORIGINAL)
        def derivatives(t, y):
            vel = y[3:6]
            f = forces.calculate(t, y)
            acc = f / mass
            return np.concatenate([vel, acc])
        
        el_rad = np.radians(req.elevation_deg)
        az_rad = np.radians(req.azimuth_deg)
        vx0 = v0 * np.cos(el_rad) * np.cos(az_rad)
        vy0 = v0 * np.cos(el_rad) * np.sin(az_rad)
        vz0 = v0 * np.sin(el_rad)
        
        y0 = np.array([0.0, 0.0, 0.0, vx0, vy0, vz0])
        
        def hit_ground(t, y): return y[2]
        hit_ground.terminal = True
        hit_ground.direction = -1
        
        result = solve_ivp(derivatives, (0, 300), y0, 
                          events=hit_ground, method='DOP853',
                          rtol=1e-10, atol=1e-12, max_step=0.5)
        
        y = result.y
        t = result.t
        x_range = np.sqrt(y[0,:]**2 + y[1,:]**2)
        z_alt = -y[2,:]
        speed = np.sqrt(y[3,:]**2 + y[4,:]**2 + y[5,:]**2)
        
        # Resultados para el frontend
        impact_idx = -1
        return {
            "success": True,
            "range_m": float(x_range[impact_idx]),
            "deflection_m": float(y[1, impact_idx]),
            "time_flight_s": float(t[impact_idx]),
            "max_altitude_m": float(np.max(z_alt)),
            "impact_velocity_ms": float(speed[impact_idx]),
            "impact_mach": float(speed[impact_idx] / atmosphere.get_properties(z_alt[impact_idx])['speed_of_sound']),
            "trajectory_points": [
                {
                    "time_s": float(t[i]),
                    "range_m": float(x_range[i]),
                    "altitude_m": float(z_alt[i]),
                    "speed_ms": float(speed[i]),
                    "deflection_m": float(y[1, i])
                }
                for i in range(0, len(t), max(1, len(t) // 200))
            ]
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/api/drev/munitions")
async def list_munitions():
    """Lista todos los proyectiles disponibles"""
    projectiles = []
    for f in os.listdir(MUNITIONS_DIR):
        if f.endswith('.json'):
            with open(os.path.join(MUNITIONS_DIR, f)) as fp:
                data = json.load(fp)
                projectiles.append({
                    "file": f,
                    "name": data.get("name", f),
                    "caliber": data.get("caliber"),
                    "charges": list(data.get("charges", {}).keys()),
                    "drag_function": data.get("drag_function", "G1")
                })
    return projectiles

@app.post("/api/drev/firing-table")
async def generate_firing_table(req: SimulationRequest):
    """Genera tabla de tiro completa usando tu FiringTableGenerator"""
    with open(os.path.join(MUNITIONS_DIR, req.projectile_file)) as f:
        proj = json.load(f)
    
    atmosphere = MolecularAir()
    atmosphere.set_temperature_offset(req.temp_offset)
    
    generator = FiringTableGenerator(proj, atmosphere, 
        latitude=req.latitude_deg,
        wind_speed=req.wind_speed,
        wind_direction=req.wind_direction)
    
    ranges = list(range(1000, 25000, 500))
    table = []
    for target in ranges:
        elev = generator.find_elevation_for_range(target, req.azimuth_deg)
        if elev:
            result = generator.simulate_trajectory(elev, req.azimuth_deg)
            table.append({
                "range_m": round(result['range']),
                "qe_mils": round(elev * (6400/360), 1),

                "max_alt_m": round(result['max_alt'], 1),
                "deflection_m": round(result['deflection'], 2),
                "impact_vel_ms": round(result['impact_vel'], 1)
            })
    
    return {"projectile": proj.get("name"), "table": table}

sudo -u postgres psql -c "CREATE DATABASE ares_os OWNER ares;"

[Install]

[Install]
WantedBy=multi-user.target

MUMBLEEOF

NGINXEOF
